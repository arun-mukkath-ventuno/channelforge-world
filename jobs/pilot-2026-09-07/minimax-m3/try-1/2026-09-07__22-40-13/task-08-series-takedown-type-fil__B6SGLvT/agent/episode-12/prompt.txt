"""Rights evaluation engine (PRD 3.0 §8, ADR-5).

A pure function over (applicable policies, context): deterministic, no I/O, so it is reused
identically at schedule generation, validation, publish, EPG generation, preflight, pre-playout,
and distributor eligibility (§8.2). Precedence is **most-restrictive-wins**: any applicable policy
that denies blocks the air, and a broader grant can never weaken a restriction (§8.1). Relaxation
happens only through an explicit, permissioned override (a later increment).

Run-count limits (§8.4) are enforced against qualifying as-run outcomes; they depend on the
reconciliation ID chain and are wired in a later increment, not here.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime

from app.clock import as_utc
from app.models import Asset, RightsPolicyVersion
from app.services.rights import covers_window


@dataclass(frozen=True)
class RightsContext:
    """The air a policy is evaluated against."""

    air_start: datetime
    air_end: datetime
    territory: str | None = None
    partner: str | None = None
    language: str | None = None
    is_avod: bool = True
    is_linear_fast: bool = True


@dataclass(frozen=True)
class ApplicablePolicy:
    """A policy version made applicable to an asset by an assignment, with its scope/priority."""

    version: RightsPolicyVersion
    scope_type: str = "asset"
    priority: int = 0


@dataclass(frozen=True)
class RightsReason:
    code: str
    message: str
    policy_version_id: str | None = None


@dataclass(frozen=True)
class RightsDecision:
    allowed: bool
    reasons: list[RightsReason] = field(default_factory=list)
    policy_version_ids: list[str] = field(default_factory=list)


def _overlaps(a_start: datetime, a_end: datetime, b_start: datetime, b_end: datetime) -> bool:
    return a_start < b_end and b_start < a_end


def _deny_reasons_for_version(
    v: RightsPolicyVersion, ctx: RightsContext
) -> list[RightsReason]:
    vid = str(v.id)
    reasons: list[RightsReason] = []

    # Validity window must fully cover the air window.
    if v.valid_start is not None and ctx.air_start < as_utc(v.valid_start):
        reasons.append(RightsReason("rights_not_yet_valid",
                                    "Policy is not yet valid for this air time.", vid))
    if v.valid_end is not None and ctx.air_end > as_utc(v.valid_end):
        reasons.append(RightsReason("rights_expired",
                                    "Policy has expired before this air window ends.", vid))

    # Territory.
    if ctx.territory is not None:
        if v.included_territories and ctx.territory not in v.included_territories:
            reasons.append(RightsReason("territory_not_included",
                                        f"Territory {ctx.territory} is not in the grant.", vid))
        if v.excluded_territories and ctx.territory in v.excluded_territories:
            reasons.append(RightsReason("territory_excluded",
                                        f"Territory {ctx.territory} is excluded.", vid))

    # Distribution partner.
    if ctx.partner is not None:
        if v.included_partners and ctx.partner not in v.included_partners:
            reasons.append(RightsReason("partner_not_included",
                                        f"Partner {ctx.partner} is not in the grant.", vid))
        if v.excluded_partners and ctx.partner in v.excluded_partners:
            reasons.append(RightsReason("partner_excluded",
                                        f"Partner {ctx.partner} is excluded.", vid))

    # Use permissions.
    if ctx.is_linear_fast and not v.allow_linear_fast:
        reasons.append(RightsReason("linear_fast_not_permitted",
                                    "Linear/FAST use is not permitted.", vid))
    if ctx.is_avod and not v.allow_avod:
        reasons.append(RightsReason("avod_not_permitted",
                                    "AVOD monetization is not permitted.", vid))

    # Language.
    if ctx.language is not None and v.allowed_languages and ctx.language not in v.allowed_languages:
        reasons.append(RightsReason("language_not_permitted",
                                    f"Language {ctx.language} is not permitted.", vid))

    # Blackout windows.
    for window in v.blackout_windows or []:
        try:
            b_start = as_utc(datetime.fromisoformat(window["start"]))
            b_end = as_utc(datetime.fromisoformat(window["end"]))
        except (KeyError, ValueError, TypeError):
            continue
        if _overlaps(ctx.air_start, ctx.air_end, b_start, b_end):
            reasons.append(RightsReason("blackout",
                                        "Air window overlaps a blackout period.", vid))
    return reasons


def evaluate(
    policies: list[ApplicablePolicy], ctx: RightsContext, *, asset: Asset | None = None
) -> RightsDecision:
    """Decide whether the air is permitted. Any applicable policy that denies blocks it.

    With no applicable policy, fall back to the asset's legacy single rights window
    (``covers_window``) so 1.x behaviour is preserved during migration (ADR-4/ADR-5).
    """
    if not policies:
        if asset is not None and not covers_window(asset, start=ctx.air_start, end=ctx.air_end):
            return RightsDecision(
                allowed=False,
                reasons=[RightsReason("asset_rights_window",
                                      "Asset's rights window does not cover the air window.")],
            )
        return RightsDecision(allowed=True)

    reasons: list[RightsReason] = []
    version_ids: list[str] = []
    for applicable in policies:
        version_ids.append(str(applicable.version.id))
        reasons.extend(_deny_reasons_for_version(applicable.version, ctx))

    return RightsDecision(allowed=not reasons, reasons=reasons, policy_version_ids=version_ids)
