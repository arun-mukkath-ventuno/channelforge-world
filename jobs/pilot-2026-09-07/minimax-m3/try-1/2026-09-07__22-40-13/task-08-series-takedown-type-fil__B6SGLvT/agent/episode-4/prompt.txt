"""Versioned rights policies and their assignments (PRD 3.0 §8, ADR-5).

A ``RightsPolicy`` is a logical grant; its terms live in immutable ``RightsPolicyVersion`` rows so
that historical evaluations can retain the exact version they used (§6.2). A ``RightsAssignment``
attaches a policy to an asset / series / collection / channel / distribution agreement.
"""

from __future__ import annotations

import uuid
from datetime import datetime

from sqlalchemy import (
    JSON,
    Boolean,
    DateTime,
    ForeignKey,
    Integer,
    String,
    Text,
    UniqueConstraint,
    Uuid,
)
from sqlalchemy.orm import Mapped, mapped_column

from app.models.base import Base, TimestampMixin, UUIDMixin


class RightsPolicy(UUIDMixin, TimestampMixin, Base):
    """A logical, versioned rights grant. Terms live in RightsPolicyVersion rows."""

    __tablename__ = "rights_policies"

    organization_id: Mapped[uuid.UUID] = mapped_column(Uuid, index=True, nullable=False)
    name: Mapped[str] = mapped_column(String(300), nullable=False)
    contract_reference: Mapped[str | None] = mapped_column(String(300), nullable=True)


class RightsPolicyVersion(UUIDMixin, TimestampMixin, Base):
    """One immutable version of a policy's terms (§8.1). Superseding creates a new version."""

    __tablename__ = "rights_policy_versions"
    __table_args__ = (
        UniqueConstraint("policy_id", "version_number", name="uq_rights_policy_version"),
    )

    organization_id: Mapped[uuid.UUID] = mapped_column(Uuid, index=True, nullable=False)
    policy_id: Mapped[uuid.UUID] = mapped_column(
        Uuid, ForeignKey("rights_policies.id", ondelete="CASCADE"), index=True, nullable=False
    )
    version_number: Mapped[int] = mapped_column(Integer, nullable=False)

    valid_start: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
    valid_end: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
    # Territory / partner allow+deny lists. Empty included list = no inclusion restriction.
    included_territories: Mapped[list | None] = mapped_column(JSON, nullable=True)
    excluded_territories: Mapped[list | None] = mapped_column(JSON, nullable=True)
    included_partners: Mapped[list | None] = mapped_column(JSON, nullable=True)
    excluded_partners: Mapped[list | None] = mapped_column(JSON, nullable=True)

    allow_linear_fast: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)
    allow_avod: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)
    allowed_languages: Mapped[list | None] = mapped_column(JSON, nullable=True)
    device_restrictions: Mapped[list | None] = mapped_column(JSON, nullable=True)
    # Blackout windows: [{"start": iso, "end": iso}, ...]
    blackout_windows: Mapped[list | None] = mapped_column(JSON, nullable=True)
    # Run limits (enforced against qualifying as-run outcomes, §8.4; wired in a later increment).
    max_runs: Mapped[int | None] = mapped_column(Integer, nullable=True)
    max_runs_period_days: Mapped[int | None] = mapped_column(Integer, nullable=True)

    exclusivity_note: Mapped[str | None] = mapped_column(String(500), nullable=True)
    notes: Mapped[str | None] = mapped_column(Text, nullable=True)


class RightsAssignment(UUIDMixin, TimestampMixin, Base):
    """Attaches a policy to a scope (asset/series/collection/channel/agreement) (§8, ADR-5).

    Precedence: assignments are ordered by scope specificity then ``priority``; a more restrictive
    rule always wins (a broader, lower-priority grant can never weaken a restriction).
    """

    __tablename__ = "rights_assignments"

    organization_id: Mapped[uuid.UUID] = mapped_column(Uuid, index=True, nullable=False)
    policy_id: Mapped[uuid.UUID] = mapped_column(
        Uuid, ForeignKey("rights_policies.id", ondelete="CASCADE"), index=True, nullable=False
    )
    # One of: asset | series | collection | channel | agreement
    scope_type: Mapped[str] = mapped_column(String(20), nullable=False)
    scope_id: Mapped[uuid.UUID] = mapped_column(Uuid, index=True, nullable=False)
    priority: Mapped[int] = mapped_column(Integer, default=0, nullable=False)
    enabled: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)


class RightsTakedown(UUIDMixin, TimestampMixin, Base):
    """An emergency takedown on an asset or series (PRD 3.0 §8.3, ADR-5).

    A non-destructive override: while ``status='active'`` the target is blocked at every eligibility
    decision point (new scheduling, EPG inclusion, distributor eligibility) without rewriting any
    immutable published schedule version. Lifting restores eligibility.
    """

    __tablename__ = "rights_takedowns"

    organization_id: Mapped[uuid.UUID] = mapped_column(Uuid, index=True, nullable=False)
    scope_type: Mapped[str] = mapped_column(String(20), nullable=False)  # asset | series
    scope_id: Mapped[uuid.UUID] = mapped_column(Uuid, index=True, nullable=False)
    reason: Mapped[str] = mapped_column(Text, nullable=False)
    status: Mapped[str] = mapped_column(String(20), default="active", nullable=False)
    actor_user_id: Mapped[uuid.UUID | None] = mapped_column(Uuid, nullable=True)
    lifted_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
    lifted_by: Mapped[uuid.UUID | None] = mapped_column(Uuid, nullable=True)
