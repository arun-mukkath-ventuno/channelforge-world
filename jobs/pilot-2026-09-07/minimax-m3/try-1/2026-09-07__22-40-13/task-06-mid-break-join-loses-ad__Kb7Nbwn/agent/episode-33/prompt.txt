import { describe, it, expect } from "vitest";
import { MemoryPodCache, type CachedPod } from "./pod-cache.js";

const ENTRY: CachedPod = {
  sessionId: "sess_1",
  opportunityId: "opp_1",
  channelId: "channel_food",
  breakDuration: 60,
  pod: [
    { type: "ad", creativeId: "cr_a", campaignId: "camp_a", duration: 30, position: 0, startOffset: 0, endOffset: 30 },
    { type: "slate", duration: 30, position: 1, startOffset: 30, endOffset: 60 },
  ],
};

describe("PodCache (control→data plane bridge, PRD §48)", () => {
  it("stores and retrieves a pod by (session, opportunity)", async () => {
    const cache = new MemoryPodCache();
    await cache.set(ENTRY);
    const got = await cache.get("sess_1", "opp_1");
    expect(got).toEqual(ENTRY);
  });

  it("isolates pods by opportunity", async () => {
    const cache = new MemoryPodCache();
    await cache.set(ENTRY);
    expect(await cache.get("sess_1", "opp_other")).toBeNull();
  });

  it("deletes a pod", async () => {
    const cache = new MemoryPodCache();
    await cache.set(ENTRY);
    await cache.del("sess_1", "opp_1");
    expect(await cache.get("sess_1", "opp_1")).toBeNull();
  });
});
