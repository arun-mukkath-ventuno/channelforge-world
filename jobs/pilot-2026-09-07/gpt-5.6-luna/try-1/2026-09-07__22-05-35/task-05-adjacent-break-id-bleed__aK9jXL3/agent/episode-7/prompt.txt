     2	// avails, and the *resolved* pod media for each avail, replace each avail's
     3	// CUE-OUT…CUE-IN region with the pod's segments (real creative segments for ads,
     4	// synthesized for slate). Avails without resolved media are left as origin (their
     5	// ChannelForge slate plays) — playback survives ad failure (§42).
     6	//
     7	// Pure and synchronous by design: no DB, no transcode, no network. Segment media
     8	// is resolved beforehand (see media.ts) so this stays a lean string transform.
     9	
    10	import type { Avail } from "./avails.js";
    11	import type { RenderedPodItem } from "./media.js";
    12	
    13	export interface StitchOptions {
    14	  /** Emit EXT-X-DATERANGE ad metadata for observability (PRD §25). Default true. */
    15	  emitAdMetadata?: boolean;
    16	}
    17	
    18	export interface StitchResult {
    19	  manifest: string;
    20	  /** Opportunity ids that were stitched (had resolved media). */
    21	  stitched: string[];
    22	}
    23	
    24	/** Replace each avail that has resolved pod media; leave the rest untouched. */
    25	export function stitchManifest(
    26	  origin: string,
    27	  avails: Avail[],
    28	  rendered: Map<string, RenderedPodItem[]>,
    29	  opts: StitchOptions = {},
    30	): StitchResult {
    31	  const emitMeta = opts.emitAdMetadata ?? true;
    32	  const lines = origin.split("\n");
    33	
    34	  const availByStart = new Map<number, Avail>();
    35	  for (const a of avails) availByStart.set(a.startLine, a);
    36	
    37	  const out: string[] = [];
    38	  const stitched: string[] = [];
    39	
    40	  for (let i = 0; i < lines.length; i++) {
    41	    const avail = availByStart.get(i);
    42	    const pod = avail ? rendered.get(avail.opportunityId) : undefined;
    43	
    44	    if (avail && pod && pod.length > 0) {
    45	      out.push(...renderPod(pod, emitMeta, avail.programDateTime));
    46	      stitched.push(avail.opportunityId);
    47	      i = avail.endLine; // skip the origin avail region (CUE-OUT…CUE-IN inclusive)
    48	      continue;
    49	    }
    50	    out.push(lines[i]!);
    51	  }
    52	
    53	  return { manifest: out.join("\n"), stitched };
    54	}

    56	function renderPod(pod: RenderedPodItem[], emitMeta: boolean, breakStart?: string): string[] {
    57	  const out: string[] = [];
