// When a viewer reaches an avail with no cached pod, the data plane asks the
// control plane's Ad Decision Service for one (PRD §15). The control plane
// persists the decision and writes the pod to the cache, so subsequent playlist
// refreshes for this (session, opportunity) are pure cache reads on the hot path.

import type { PodItem } from "@ssai/core";
import type { Avail } from "./avails.js";

export interface DecideRequest {
  sessionId: string;
  channelId: string;
  avail: Avail;
}

export interface PodDecider {
  decide(req: DecideRequest): Promise<PodItem[] | null>;
}

/** Posts to the control plane's /v1/ad-decision. */
export class HttpPodDecider implements PodDecider {
  constructor(
    private readonly controlPlaneUrl: string,
    private readonly fetchImpl: typeof fetch = fetch,
    private readonly timeoutMs = 200,
  ) {}

  async decide({ sessionId, channelId, avail }: DecideRequest): Promise<PodItem[] | null> {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), this.timeoutMs);
    try {
      const res = await this.fetchImpl(`${this.controlPlaneUrl}/v1/ad-decision`, {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          session_id: sessionId,
          channel_id: channelId,
          opportunity_id: avail.opportunityId,
          duration: avail.duration,
          ...(avail.externalBreakId === undefined
            ? {}
            : { external_break_id: avail.externalBreakId }),
        }),
        signal: controller.signal,
      });
      if (!res.ok) return null;
      const decision = (await res.json()) as { pod?: PodItem[] };
      return decision.pod ?? null;
    } catch {
      // ADS unavailable → no pod → origin slate plays (PRD §37).
      return null;
    } finally {
      clearTimeout(timer);
    }
  }
}

// Playback session repository (PRD §11, §12, §44).

import type { Queryable } from "../db.js";
import { toIso } from "../map.js";

export interface PlaybackSession {
  id: string;
  channelId: string;
  createdAt: string;
  lastActivityAt: string;
  deviceType: string | null;
  country: string | null;
  anonymousId: string | null;
}

interface SessionRow {
  id: string;
  channel_id: string;
  created_at: unknown;
  last_activity_at: unknown;
  device_type: string | null;
  country: string | null;
  anonymous_id: string | null;
}

export interface SessionInput {
  id: string;
  channelId: string;
  deviceType?: string | null;
  country?: string | null;
  anonymousId?: string | null;
  ipHash?: string | null;
  userAgent?: string | null;
}

/** Create the session if absent; idempotent so ad-decision can ensure it exists. */
export async function ensureSession(db: Queryable, input: SessionInput): Promise<void> {
  await db.query(
    `INSERT INTO playback_sessions (id, channel_id, device_type, country, anonymous_id, ip_hash, user_agent)
     VALUES ($1, $2, $3, $4, $5, $6, $7)
     ON CONFLICT (id) DO NOTHING`,
    [
      input.id,
      input.channelId,
      input.deviceType ?? null,
      input.country ?? null,
      input.anonymousId ?? null,
      input.ipHash ?? null,
      input.userAgent ?? null,
    ],
  );
}

export async function touchSession(db: Queryable, id: string): Promise<void> {
  await db.query("UPDATE playback_sessions SET last_activity_at = now() WHERE id = $1", [id]);
}

function toSession(r: SessionRow): PlaybackSession {
  return {
    id: r.id,
    channelId: r.channel_id,
    createdAt: toIso(r.created_at),
    lastActivityAt: toIso(r.last_activity_at),
    deviceType: r.device_type,
    country: r.country,
    anonymousId: r.anonymous_id,
  };
}

export async function getSession(db: Queryable, id: string): Promise<PlaybackSession | null> {
  const rows = (await db.query<SessionRow>("SELECT * FROM playback_sessions WHERE id = $1", [id])).rows;
  return rows[0] ? toSession(rows[0]) : null;
}

export async function listSessions(db: Queryable, limit = 100): Promise<PlaybackSession[]> {
  const rows = (
    await db.query<SessionRow>("SELECT * FROM playback_sessions ORDER BY last_activity_at DESC LIMIT $1", [limit])
  ).rows;
  return rows.map(toSession);
}
