// SSAI (server-side ad insertion) client — the AD leg of the Stream Telemetry Blueprint.
//
// For channels with an `ssaiChannel`, the player requests a session from the ad server, plays the
// returned stitched `playback_url` (ChannelForge content + spliced ads), and fires per-ad quartile
// beacons. This is SEPARATE from the audience telemetry in `telemetry-store` / `/api/fast/events`
// (concurrents, QoE) — that stays untouched; ad events go to the SSAI's `/v1/events`.

export const SSAI_BASE = (
  process.env.NEXT_PUBLIC_SSAI_BASE || "https://ssai.ventunotech.com"
).replace(/\/$/, "");

export interface SsaiSession {
  sessionId: string;
  playbackUrl: string;
}

/** Create a playback session and get the stitched manifest URL for an SSAI channel. */
export async function createSsaiSession(ssaiChannel: string): Promise<SsaiSession | null> {
  try {
    const res = await fetch(`${SSAI_BASE}/v1/sessions`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ channel_id: ssaiChannel }),
    });
    if (!res.ok) return null;
    const d = await res.json();
    if (!d?.session_id || !d?.playback_url) return null;
    return { sessionId: String(d.session_id), playbackUrl: String(d.playback_url) };
  } catch {
    return null;
  }
}

export type AdBeacon =
  | "ad_impression"
  | "ad_start"
  | "ad_first_quartile"
  | "ad_midpoint"
  | "ad_third_quartile"
  | "ad_complete";

/** Fire one ad tracking beacon to the SSAI. Best-effort; failures never disrupt playback. */
export function sendAdBeacon(sessionId: string, event: AdBeacon, adId: string): void {
  const body = JSON.stringify({ event, session_id: sessionId, ad_id: adId });
  try {
    if (typeof navigator !== "undefined" && navigator.sendBeacon) {
      navigator.sendBeacon(`${SSAI_BASE}/v1/events`, new Blob([body], { type: "application/json" }));
    } else {
      void fetch(`${SSAI_BASE}/v1/events`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body,
        keepalive: true,
      });
    }
  } catch {
    /* best-effort */
  }
}

export interface AdWindow {
  adId: string;
  start: number; // epoch ms
  end: number; // epoch ms
}

/** Parse ad avails from a stitched media playlist's EXT-X-DATERANGE lines (CLASS com.fastworld.ad). */
export function parseAdWindows(manifest: string): AdWindow[] {
  const out: AdWindow[] = [];
  for (const line of manifest.split("\n")) {
    if (!line.startsWith("#EXT-X-DATERANGE:") || !line.includes('CLASS="com.fastworld.ad"')) continue;
    const attr = (k: string): string | undefined => line.match(new RegExp(`${k}="?([^",]+)"?`))?.[1];
    const startDate = attr("START-DATE");
    const dur = attr("DURATION");
    const adId = attr("X-AD-ID") || attr("ID");
    if (!startDate || !dur || !adId) continue;
    const start = Date.parse(startDate);
    const seconds = Number(dur);
    if (Number.isNaN(start) || Number.isNaN(seconds)) continue;
    out.push({ adId, start, end: start + seconds * 1000 });
  }
  return out;
}

export interface ScteAvail {
  id: string;
  start: number; // epoch ms — break start (absolute, from START-DATE)
  end: number; // epoch ms
  durationSec: number;
}

/**
 * Parse SCTE-35 ad avails from a ChannelForge media playlist. These are the ad
 * OPPORTUNITIES the packager signals (where SSAI may splice), distinct from
 * `parseAdWindows` which reads the ads actually stitched in. Reads
 * `#EXT-X-DATERANGE` lines carrying `SCTE35-OUT` with an absolute `START-DATE`
 * and `PLANNED-DURATION`.
 */
export function parseScteAvails(manifest: string): ScteAvail[] {
  const out: ScteAvail[] = [];
  for (const line of manifest.split("\n")) {
    if (!line.startsWith("#EXT-X-DATERANGE:")) continue;
    if (!/SCTE35-OUT/i.test(line)) continue;
    const attr = (k: string): string | undefined =>
      line.match(new RegExp(`${k}="?([^",]+)"?`))?.[1];
    const startDate = attr("START-DATE");
    const dur = attr("PLANNED-DURATION") || attr("DURATION");
    const id = attr("ID") || "brk";
    if (!startDate || !dur) continue;
    const start = Date.parse(startDate);
    const seconds = Number(dur);
    if (Number.isNaN(start) || Number.isNaN(seconds)) continue;
    out.push({ id, start, end: start + seconds * 1000, durationSec: seconds });
  }
  return out;
}
