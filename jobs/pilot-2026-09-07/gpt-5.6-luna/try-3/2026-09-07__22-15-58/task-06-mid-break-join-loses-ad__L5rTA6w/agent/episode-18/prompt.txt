const MEDIA_SEQUENCE = /^#EXT-X-MEDIA-SEQUENCE:(\d+)/;
const EXTINF = /^#EXTINF:([\d.]+)/;
const PROGRAM_DATE_TIME = /^#EXT-X-PROGRAM-DATE-TIME:(.+)/;
const CUE_OUT_CONT = /^#EXT-X-CUE-OUT-CONT(?::(.*))?$/;

function continuationAttrs(value: string): { elapsed?: number; duration?: number } {
  const attr = (name: string): number | undefined => {
    const match = value.match(new RegExp(name + "=\\s*(\\d+(?:\\.\\d+)?)", "i"));
    return match ? Number(match[1]) : undefined;
  };
  return { elapsed: attr("ElapsedTime"), duration: attr("Duration") };
}
