  });

  it("rejects a wrong password, accepts the right one and gates on the cookie", async () => {
    const app = buildWithAuth();
    expect((await app.inject({ method: "POST", url: "/admin/api/login", payload: { user: "ops", password: "nope" } })).statusCode).toBe(401);

    const ok = await app.inject({ method: "POST", url: "/admin/api/login", payload: { user: "ops", password: "s3cret" } });
    expect(ok.statusCode).toBe(200);
    expect(ok.json()).toEqual({ user: "ops" });
    const cookie = ok.cookies.find((c) => c.name === "ssai_admin");
    expect(cookie).toBeTruthy();

    const header = { cookie: `ssai_admin=${cookie!.value}` };
    expect((await app.inject({ method: "GET", url: "/admin/api/me", headers: header })).json()).toEqual({ user: "ops" });
    expect((await app.inject({ method: "GET", url: "/v1/channels", headers: header })).statusCode).toBe(200);
