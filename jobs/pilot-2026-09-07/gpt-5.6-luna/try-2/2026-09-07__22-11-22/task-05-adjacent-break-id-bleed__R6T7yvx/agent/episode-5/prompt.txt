     1	// Absolutize an origin media playlist's RELATIVE URIs against the origin playlist URL.
     2	//
     3	// When the SSAI wraps a real EXTERNAL ChannelForge origin, the stitched manifest is served from
     4	// the SSAI host — so an origin's relative content URI like `index0.ts` would otherwise resolve
     5	// against the SSAI (`…/v1/hls/{token}/{channel}/index0.ts`) and 404. Rewriting content URIs to
     6	// absolute origin URLs (`https://forge…/hls/{uuid}/index0.ts`) lets the player pull content
     7	// straight from ChannelForge, while ad segments the stitcher inserts stay absolute SSAI paths.
     8	// For the co-located fixture origin (relative URIs that already resolve) this is a harmless no-op
     9	// difference, so it's safe to apply unconditionally.
    10	
    11	function absolutize(uri: string, base: URL): string {
    12	  const t = uri.trim();
    13	  if (!t || /^https?:\/\//i.test(t)) return uri; // already absolute
    14	  try {
    15	    return new URL(t, base).toString();
    16	  } catch {
    17	    return uri;
    18	  }
    19	}
    20	
    21	export function absolutizeManifest(body: string, originUrl: string): string {
    22	  let base: URL;
    23	  try {
    24	    base = new URL(originUrl);
    25	  } catch {
    26	    return body; // non-absolute origin (e.g. fixture http://host/...) still parses; bad input → passthrough
    27	  }
    28	  return body
    29	    .split("\n")
    30	    .map((line) => {
    31	      const t = line.trim();
    32	      if (!t) return line;
    33	      // A bare segment URI line (not a tag).
    34	      if (!t.startsWith("#")) return absolutize(line, base);
    35	      // Tags that carry a URI="..." attribute (EXT-X-MAP, EXT-X-KEY, …).
    36	      return line.replace(/URI="([^"]+)"/g, (_m, u) => `URI="${absolutize(u, base)}"`);
    37	    })
    38	    .join("\n");
    39	}
