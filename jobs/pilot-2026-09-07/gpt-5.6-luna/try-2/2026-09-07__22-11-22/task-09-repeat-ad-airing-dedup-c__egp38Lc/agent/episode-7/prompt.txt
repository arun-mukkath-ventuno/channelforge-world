   100	  const out: ScteAvail[] = [];
   101	  for (const line of manifest.split("\n")) {
   102	    if (!line.startsWith("#EXT-X-DATERANGE:")) continue;
   103	    if (!/SCTE35-OUT/i.test(line)) continue;
   104	    const attr = (k: string): string | undefined =>
   105	      line.match(new RegExp(`${k}="?([^",]+)"?`))?.[1];
   106	    const startDate = attr("START-DATE");
   107	    const dur = attr("PLANNED-DURATION") || attr("DURATION");
   108	    const id = attr("ID") || "brk";

   112	    if (Number.isNaN(start) || Number.isNaN(seconds)) continue;
   113	    out.push({ id, start, end: start + seconds * 1000, durationSec: seconds });
