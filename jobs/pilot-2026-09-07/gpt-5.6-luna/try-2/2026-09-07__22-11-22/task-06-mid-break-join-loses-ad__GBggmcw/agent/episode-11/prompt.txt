function parseCueOutDuration(value: string): number | null {
  if (value.length === 0) return null;
  if (SECONDS.test(value)) {
    const n = Number(value);

/** Parse `KEY=value` (optionally quoted) out of a comma-separated attribute list. */
function parseAttr(attrs: string, key: string): number | null {
  const re = new RegExp(`${key}=("?)(-?\\d+(?:\\.\\d+)?)\\1`, "i");
  const m = attrs.match(re);
  if (!m) return null;
  const n = Number(m[2]);
  return Number.isFinite(n) && n >= 0 ? n : null;
}
