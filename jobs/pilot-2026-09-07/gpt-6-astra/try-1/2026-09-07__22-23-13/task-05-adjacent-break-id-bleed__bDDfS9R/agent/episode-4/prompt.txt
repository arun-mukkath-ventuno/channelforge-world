// Avail detection over an origin media playlist (PRD §9, §22, §41).
//
// Walks the playlist, finds each CUE-OUT…CUE-IN region, and assigns a
// *deterministic* opportunity id derived from the channel + the media sequence
// of the avail's first segment. Determinism matters: every viewer's data-plane
// request and the decisioning path must agree on the same opportunity id for a
// given break without coordinating, so the cached pod is found on the hot path.

import { parseMarkerLine } from "@ssai/core";

export interface Avail {
  opportunityId: string;
  channelId: string;
  /** The producer's stable break id, from the CUE-OUT's accompanying DATERANGE ID (ChannelForge
   *  threads its break_id there). Undefined when the origin carried no DATERANGE ID. This is what
   *  lets ChannelForge reconcile a delivery back to the exact planned break (its opportunityId,
   *  by contrast, is a media-sequence-derived id independent of the producer). */
  externalBreakId?: string;
  /** Break duration: the marker's declared duration, or the summed segment time. */
  duration: number;
  /** Declared duration from the marker, if it carried a valid one. */
  declaredDuration?: number;
  /** Actual duration summed from the segments inside the avail. */
  measuredDuration: number;
  /** Media sequence number of the first segment inside the avail. */
  startSequence: number;
  programDateTime?: string;
  /** Line index of the CUE-OUT marker. */
  startLine: number;
  /** Line index of the CUE-IN marker. */
  endLine: number;
}

const MEDIA_SEQUENCE = /^#EXT-X-MEDIA-SEQUENCE:(\d+)/;
const EXTINF = /^#EXTINF:([\d.]+)/;
const PROGRAM_DATE_TIME = /^#EXT-X-PROGRAM-DATE-TIME:(.+)/;

  // one break's id never leaks onto the next. ChannelForge emits the CUE-OUT and its DATERANGE-OUT

// When a viewer reaches an avail with no cached pod, the data plane asks the
// control plane's Ad Decision Service for one (PRD §15). The control plane
// persists the decision and writes the pod to the cache, so subsequent playlist
// refreshes for this (session, opportunity) are pure cache reads on the hot path.

import type { PodItem } from "@ssai/core";
import type { Avail } from "./avails.js";

export interface DecideRequest {
  sessionId: string;
  channelId: string;
  avail: Avail;
}

export interface PodDecider {
  decide(req: DecideRequest): Promise<PodItem[] | null>;
}

/** Posts to the control plane's /v1/ad-decision. */
export class HttpPodDecider implements PodDecider {
  constructor(
    private readonly controlPlaneUrl: string,
    private readonly fetchImpl: typeof fetch = fetch,
    private readonly timeoutMs = 200,
  ) {}

  async decide({ sessionId, channelId, avail }: DecideRequest): Promise<PodItem[] | null> {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), this.timeoutMs);
    try {
      const res = await this.fetchImpl(`${this.controlPlaneUrl}/v1/ad-decision`, {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          session_id: sessionId,
          channel_id: channelId,
          opportunity_id: avail.opportunityId,
          duration: avail.duration,
          ...(avail.externalBreakId === undefined
            ? {}
            : { external_break_id: avail.externalBreakId }),
        }),
        signal: controller.signal,
      });
      if (!res.ok) return null;
      const decision = (await res.json()) as { pod?: PodItem[] };
      return decision.pod ?? null;
    } catch {
      // ADS unavailable → no pod → origin slate plays (PRD §37).
      return null;
    } finally {
      clearTimeout(timer);
    }
  }
}
