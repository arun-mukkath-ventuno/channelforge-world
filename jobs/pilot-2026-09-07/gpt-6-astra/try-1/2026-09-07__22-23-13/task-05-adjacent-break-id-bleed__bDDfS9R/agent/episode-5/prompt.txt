// HLS ad-marker parsing (PRD §9, §22, §41).
//
// Supports the MVP-preferred form and common interoperability variants:
//   #EXT-X-CUE-OUT:60
//   #EXT-X-CUE-OUT:DURATION=60
//   #EXT-X-CUE-OUT-CONT:...        (continuation, ignored as a new avail)
//   #EXT-X-CUE-IN
//   #EXT-X-DATERANGE:...,DURATION=60,SCTE35-OUT=...
//
// This parser is deliberately tolerant: an unparseable duration does not throw,
// it yields a marker flagged invalid so the caller can apply PRD §41 fallback
// (infer the boundary from CUE-IN, otherwise preserve origin).

export interface ParsedMarker {
  kind: "cue_out" | "cue_in" | "daterange";
  /** Break duration in seconds, when the marker carries one. */
  duration?: number;
  /** Line index within the manifest where the marker was found. */
  lineIndex: number;
  /** DATERANGE id / raw payload, retained for debugging (PRD §46). */
  raw: string;
  /** DATERANGE ID attribute, when present. ChannelForge threads its stable break_id here
   *  (`#EXT-X-DATERANGE:ID="BRK-…"`), so downstream reconciliation can key on it. */
  id?: string;
  /** True when a duration was expected but could not be parsed (PRD §41). */
  invalidDuration?: boolean;
}

const SECONDS = /^-?\d+(\.\d+)?$/;

/** Parse a single manifest line into a marker, or null if it is not one. */
export function parseMarkerLine(line: string, lineIndex: number): ParsedMarker | null {
  const trimmed = line.trim();

  // CUE-IN: no attributes.
  if (trimmed === "#EXT-X-CUE-IN") {
    return { kind: "cue_in", lineIndex, raw: trimmed };
  }

  // CUE-OUT: value may be a bare number or DURATION=<n>.
  // Note: CUE-OUT-CONT is a continuation of an in-progress avail, not a new one.
  if (trimmed.startsWith("#EXT-X-CUE-OUT") && !trimmed.startsWith("#EXT-X-CUE-OUT-CONT")) {
    const value = trimmed.slice("#EXT-X-CUE-OUT".length).replace(/^:/, "").trim();
    const duration = parseCueOutDuration(value);
    return {
      kind: "cue_out",
      lineIndex,
      raw: trimmed,
      ...(duration === null ? { invalidDuration: value.length > 0 } : { duration }),
    };
  }

  // DATERANGE: pull DURATION and PLANNED-DURATION attributes if present.
  if (trimmed.startsWith("#EXT-X-DATERANGE")) {
    const attrs = trimmed.slice("#EXT-X-DATERANGE".length).replace(/^:/, "");
    const duration = parseAttr(attrs, "DURATION") ?? parseAttr(attrs, "PLANNED-DURATION");
    const id = parseStringAttr(attrs, "ID");
    return {
      kind: "daterange",
      lineIndex,
      raw: trimmed,
      ...(duration === null ? {} : { duration }),
      ...(id === null ? {} : { id }),
    };
  }

  return null;
}

/** Accepts `60`, `60.0`, `DURATION=60`, or `DURATION=60,...`. Returns null if absent/invalid. */
function parseCueOutDuration(value: string): number | null {
  if (value.length === 0) return null;
  if (SECONDS.test(value)) {
    const n = Number(value);
    return n >= 0 ? n : null;
  }
  // Attribute form: DURATION=60 (possibly with more comma-separated attrs).
  return parseAttr(value, "DURATION");
}

/** Parse `KEY=value` (optionally quoted) out of a comma-separated attribute list. */
function parseAttr(attrs: string, key: string): number | null {
  const re = new RegExp(`${key}=("?)(-?\\d+(?:\\.\\d+)?)\\1`, "i");
  const m = attrs.match(re);
  if (!m) return null;
  const n = Number(m[2]);
  return Number.isFinite(n) && n >= 0 ? n : null;
}

/** Parse a string `KEY="value"` (or bare `KEY=value`) attribute; null if absent/empty. */

import type { ObjectStore } from "./storage.js";
import { FilesystemObjectStore } from "./storage.js";

export async function createObjectStoreFromEnv(): Promise<ObjectStore> {
  if (process.env.S3_BUCKET) {
    const { createS3ObjectStore } = await import("./s3.js");
    return createS3ObjectStore({
      bucket: process.env.S3_BUCKET,
      ...(process.env.S3_ENDPOINT ? { endpoint: process.env.S3_ENDPOINT } : {}),
      ...(process.env.S3_REGION ? { region: process.env.S3_REGION } : {}),
      ...(process.env.S3_ACCESS_KEY ? { accessKeyId: process.env.S3_ACCESS_KEY } : {}),
      ...(process.env.S3_SECRET_KEY ? { secretAccessKey: process.env.S3_SECRET_KEY } : {}),
      forcePathStyle: true,

    });
  }
  return new FilesystemObjectStore(process.env.CREATIVE_STORE_ROOT ?? "./storage/creatives");
}

import pg from "pg";

const { Pool } = pg;

/** Anything that can run a parameterized query — pg.Pool, pg.PoolClient, pg-mem. */
export interface Queryable {
  query<R extends pg.QueryResultRow = pg.QueryResultRow>(
    text: string,
    params?: readonly unknown[],
  ): Promise<pg.QueryResult<R>>;
}

export type Pool = pg.Pool;

export function createPool(connectionString = process.env.DATABASE_URL): pg.Pool {
  if (!connectionString) {
    throw new Error("DATABASE_URL is not set");
  }
  return new Pool({ connectionString, max: 10 });
}

/** Run `fn` inside a transaction, rolling back on error. */
export async function withTransaction<T>(pool: pg.Pool, fn: (tx: Queryable) => Promise<T>): Promise<T> {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");
    const result = await fn(client);
    await client.query("COMMIT");
    return result;
  } catch (err) {
    await client.query("ROLLBACK");
    throw err;
  } finally {
    client.release();
  }
}
