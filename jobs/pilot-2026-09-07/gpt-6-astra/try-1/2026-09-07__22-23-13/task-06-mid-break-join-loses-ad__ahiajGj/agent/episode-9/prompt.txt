// In-memory store seeded with one prepared creative.
function storeWith(entries: Record<string, string>): ObjectStore {
  return {
    async get(key) {
      if (entries[key] === undefined) throw new Error("not found: " + key);
      return Buffer.from(entries[key]);
    },
    async put() {},
    async list() {
      return [];
    },
    url: (k) => "store://local/" + k,
  };
}

      { duration: 8.333333, uri: "cr_a_001.ts" },

  });
});

describe("resolvePodSegments (PRD §14, §42)", () => {
  const pod: PodItem[] = [
    { type: "ad", creativeId: "cr_a", campaignId: "camp_a", duration: 30, position: 0, startOffset: 0, endOffset: 30 },
    { type: "slate", duration: 15, position: 1, startOffset: 30, endOffset: 45 },

  });
