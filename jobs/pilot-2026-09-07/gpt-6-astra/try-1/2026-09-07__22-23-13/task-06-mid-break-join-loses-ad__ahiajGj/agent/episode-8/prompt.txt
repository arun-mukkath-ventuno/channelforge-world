    const body = ["#EXTM3U", "#EXTINF:2.000,", "index0.ts", "#EXTINF:2.000,", "index1.ts"].join("\n");
    const out = absolutizeManifest(body, origin);
    expect(out).toContain("https://forge.ventunotech.com/hls/a11e5310/index0.ts");
    expect(out).toContain("https://forge.ventunotech.com/hls/a11e5310/index1.ts");
    expect(out).toContain("#EXTINF:2.000,"); // tags untouched
  });

  it("leaves already-absolute URIs and ad segments alone", () => {
    const body = ["#EXTINF:2,", "https://cdn/x.ts", "#EXTINF:2,", "/v1/segments/cr_1/000.ts"].join("\n");
    const out = absolutizeManifest(body, origin);

import { describe, it, expect, vi } from "vitest";
import { OriginClient } from "./origin.js";

function okResponse(body: string) {
  return { ok: true, status: 200, text: async () => body } as unknown as Response;
}

describe("OriginClient (PRD §8, §42)", () => {
  it("fetches and caches within the TTL", async () => {
    const fetchImpl = vi.fn(async () => okResponse("#EXTM3U\n"));
    const client = new OriginClient({ baseUrl: "http://origin", ttlMs: 1000, fetchImpl });

    const a = await client.fetchMediaPlaylist("ch");
    const b = await client.fetchMediaPlaylist("ch");

    expect(a).toMatchObject({ fromCache: false, stale: false });
    expect(b).toMatchObject({ fromCache: true, stale: false });
    expect(fetchImpl).toHaveBeenCalledTimes(1); // second call served from cache
  });

  it("serves the last-good copy when the origin fails (stale)", async () => {
    let call = 0;
    const fetchImpl = vi.fn(async () => {
      call += 1;
      if (call === 1) return okResponse("#EXTM3U\nfresh\n");
      throw new Error("origin down");
    });
    const client = new OriginClient({ baseUrl: "http://origin", ttlMs: 0, fetchImpl });

    const first = await client.fetchMediaPlaylist("ch");
    const second = await client.fetchMediaPlaylist("ch"); // ttl 0 forces refetch → fails

    expect(first.stale).toBe(false);
    expect(second).toMatchObject({ body: "#EXTM3U\nfresh\n", stale: true });
  });

  it("throws when the origin fails and nothing is cached", async () => {
    const fetchImpl = vi.fn(async () => {
      throw new Error("origin down");
    });
    const client = new OriginClient({ baseUrl: "http://origin", fetchImpl });
    await expect(client.fetchMediaPlaylist("ch")).rejects.toThrow("origin down");
  });

  it("treats a non-200 as a failure", async () => {
    const fetchImpl = vi.fn(async () => ({ ok: false, status: 404, text: async () => "" }) as unknown as Response);

  });

// Pod cache (PRD §48): the control plane writes a decided pod here; the data
// plane reads it on the manifest hot path and never recomputes. Keyed by
// (session, opportunity).
//
// Two implementations behind one interface: Redis for real deployments, and an
// in-memory Map for tests and single-process dev.

import type { PodItem } from "@ssai/core";
import { Redis } from "ioredis";

export interface CachedPod {
  sessionId: string;
  opportunityId: string;
  channelId: string;
  breakDuration: number;
  pod: PodItem[];
}

export interface PodCache {
  get(sessionId: string, opportunityId: string): Promise<CachedPod | null>;
  set(entry: CachedPod, ttlSeconds?: number): Promise<void>;
  del(sessionId: string, opportunityId: string): Promise<void>;
}

function key(sessionId: string, opportunityId: string): string {
  return `pod:${sessionId}:${opportunityId}`;
}

/** Default pod TTL: long enough to outlast a break, short enough to self-clean. */
export const DEFAULT_POD_TTL_SECONDS = 900;

export class RedisPodCache implements PodCache {
  constructor(private readonly redis: Redis) {}

  async get(sessionId: string, opportunityId: string): Promise<CachedPod | null> {
    const raw = await this.redis.get(key(sessionId, opportunityId));
    return raw ? (JSON.parse(raw) as CachedPod) : null;
  }

  async set(entry: CachedPod, ttlSeconds = DEFAULT_POD_TTL_SECONDS): Promise<void> {
    await this.redis.set(key(entry.sessionId, entry.opportunityId), JSON.stringify(entry), "EX", ttlSeconds);
  }

  async del(sessionId: string, opportunityId: string): Promise<void> {
    await this.redis.del(key(sessionId, opportunityId));
  }
}

export class MemoryPodCache implements PodCache {
  private readonly store = new Map<string, CachedPod>();

  async get(sessionId: string, opportunityId: string): Promise<CachedPod | null> {
    return this.store.get(key(sessionId, opportunityId)) ?? null;
  }

  async set(entry: CachedPod): Promise<void> {
    this.store.set(key(entry.sessionId, entry.opportunityId), entry);
  }

  async del(sessionId: string, opportunityId: string): Promise<void> {
    this.store.delete(key(sessionId, opportunityId));
  }
}

export function createRedis(url = process.env.REDIS_URL): Redis {
  if (!url) throw new Error("REDIS_URL is not set");
  return new Redis(url, { maxRetriesPerRequest: 2, lazyConnect: false });
}
