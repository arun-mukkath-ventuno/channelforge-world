/** Resolve the canonical event type from an explicit type or a quartile marker. */
export function canonicalType(raw: RawEvent): string {
  const explicit = (raw.event_type ?? raw.event ?? "").toString();
  const key = explicit.toLowerCase().replace(/\s+/g, "");

  if ((key === "ad_quartile" || key === "quartile") && typeof raw.quartile === "number") {
    return QUARTILE_TO_EVENT[raw.quartile] ?? explicit;
  }
  return ALIASES[key] ?? explicit;
}

/** Deterministic id for lifecycle events so duplicates dedup; unique otherwise. */
export function deriveEventId(raw: RawEvent, eventType: string): string {
  if (raw.event_id) return raw.event_id;

  const isAdLifecycle = eventType.startsWith("ad_") && eventType !== "ad_error";
  if (isAdLifecycle && raw.session_id) {
    const basis = [raw.session_id, raw.pod_id ?? "", raw.ad_id ?? "", eventType].join("|");
    return `evt_${createHash("sha1").update(basis).digest("hex").slice(0, 24)}`;
  }
  // Non-lifecycle (errors, unknown) — let each occurrence be recorded.
  return `evt_${createHash("sha1").update(`${JSON.stringify(raw)}|${Date.now()}|${Math.random()}`).digest("hex").slice(0, 24)}`;
}

export function normalizeEvent(raw: RawEvent): NormalizedEvent | { error: string } {
  const eventType = canonicalType(raw);
  if (!eventType) return { error: "event_type is required" };

  return {
    eventId: deriveEventId(raw, eventType),
    eventType,
    sessionId: raw.session_id ?? null,
    channelId: raw.channel_id ?? null,

export interface ScteAvail {
  id: string;
  start: number; // epoch ms — break start (absolute, from START-DATE)
  end: number; // epoch ms
  durationSec: number;
}

/**
 * Parse SCTE-35 ad avails from a ChannelForge media playlist. These are the ad
 * OPPORTUNITIES the packager signals (where SSAI may splice), distinct from
 * `parseAdWindows` which reads the ads actually stitched in. Reads
 * `#EXT-X-DATERANGE` lines carrying `SCTE35-OUT` with an absolute `START-DATE`
 * and `PLANNED-DURATION`.
 */
export function parseScteAvails(manifest: string): ScteAvail[] {
  const out: ScteAvail[] = [];
  for (const line of manifest.split("\n")) {
    if (!line.startsWith("#EXT-X-DATERANGE:")) continue;
    if (!/SCTE35-OUT/i.test(line)) continue;
    const attr = (k: string): string | undefined =>
      line.match(new RegExp(`${k}="?([^",]+)"?`))?.[1];
    const startDate = attr("START-DATE");
    const dur = attr("PLANNED-DURATION") || attr("DURATION");
    const id = attr("ID") || "brk";
    if (!startDate || !dur) continue;
    const start = Date.parse(startDate);
    const seconds = Number(dur);
    if (Number.isNaN(start) || Number.isNaN(seconds)) continue;
    out.push({ id, start, end: start + seconds * 1000, durationSec: seconds });
  }
  return out;
}

  [0, "ad_start"],

/**
 * Watch the wall-clock playhead against the stitched manifest's ad windows and fire quartile
 * beacons as each ad plays. `playheadDate()` returns the current playhead as epoch ms (from
 * Shaka's getPlayheadTimeAsDate). Returns a stop() cleanup.
 */
export function startAdBeaconWatcher(opts: {
  sessionId: string;
  manifestUrl: string;
  playheadDate: () => number | null;
  refetchMs?: number;
  tickMs?: number;
}): () => void {
  let windows: AdWindow[] = [];
  const fired = new Set<string>(); // `${adId}:${start}:${beacon}`
  let stopped = false;

  async function refresh() {
    try {
      const res = await fetch(opts.manifestUrl, { cache: "no-store" });
      if (res.ok) windows = parseAdWindows(await res.text());
    } catch {
      /* keep prior windows */
    }
  }

  const tick = () => {
    if (stopped) return;
    const now = opts.playheadDate();
    if (now != null) {
      for (const w of windows) {
        if (now < w.start || now > w.end) continue;
        const progress = (now - w.start) / Math.max(1, w.end - w.start);
        for (const [threshold, beacon] of QUARTILES) {
          if (progress + 1e-6 < threshold) continue;
          const key = `${w.adId}:${w.start}:${beacon}`;
          if (fired.has(key)) continue;
          fired.add(key);
          sendAdBeacon(opts.sessionId, beacon, w.adId);
        }
      }
    }
  };

  void refresh();
  const refetch = setInterval(refresh, opts.refetchMs ?? 10_000);
  const ticker = setInterval(tick, opts.tickMs ?? 1_000);
  return () => {
    stopped = true;
    clearInterval(refetch);
    clearInterval(ticker);
  };
}

  let body: unknown;

  }
  const items: Incoming[] = Array.isArray(body) ? body : [body];

    accepted++;
