// control plane's Ad Decision Service for one (PRD §15). The control plane
// persists the decision and writes the pod to the cache, so subsequent playlist
// refreshes for this (session, opportunity) are pure cache reads on the hot path.

import type { PodItem } from "@ssai/core";
import type { Avail } from "./avails.js";

export interface DecideRequest {
  sessionId: string;
  channelId: string;
  avail: Avail;
}

export interface PodDecider {
  decide(req: DecideRequest): Promise<PodItem[] | null>;
}

/** Posts to the control plane's /v1/ad-decision. */
export class HttpPodDecider implements PodDecider {
  constructor(
    private readonly controlPlaneUrl: string,
    private readonly fetchImpl: typeof fetch = fetch,
    private readonly timeoutMs = 200,
  ) {}

  async decide({ sessionId, channelId, avail }: DecideRequest): Promise<PodItem[] | null> {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), this.timeoutMs);
    try {
      const res = await this.fetchImpl(`${this.controlPlaneUrl}/v1/ad-decision`, {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          session_id: sessionId,
          channel_id: channelId,
          opportunity_id: avail.opportunityId,
          duration: avail.duration,
          ...(avail.externalBreakId === undefined
            ? {}
            : { external_break_id: avail.externalBreakId }),
        }),
        signal: controller.signal,
      });
      if (!res.ok) return null;
      const decision = (await res.json()) as { pod?: PodItem[] };
      return decision.pod ?? null;
    } catch {
      // ADS unavailable → no pod → origin slate plays (PRD §37).
      return null;
    } finally {
      clearTimeout(timer);
    }
  }
}

// Postgres connection + a minimal Queryable abstraction so repositories work
// against both a real `pg.Pool` and the pg-mem adapter used in tests.

import pg from "pg";

const { Pool } = pg;

/** Anything that can run a parameterized query — pg.Pool, pg.PoolClient, pg-mem. */
export interface Queryable {
  query<R extends pg.QueryResultRow = pg.QueryResultRow>(
    text: string,
    params?: readonly unknown[],
  ): Promise<pg.QueryResult<R>>;
}

export type Pool = pg.Pool;

export function createPool(connectionString = process.env.DATABASE_URL): pg.Pool {
  if (!connectionString) {
    throw new Error("DATABASE_URL is not set");
  }
  return new Pool({ connectionString, max: 10 });
}

/** Run `fn` inside a transaction, rolling back on error. */
export async function withTransaction<T>(pool: pg.Pool, fn: (tx: Queryable) => Promise<T>): Promise<T> {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");
    const result = await fn(client);
    await client.query("COMMIT");
    return result;
  } catch (err) {
    await client.query("ROLLBACK");
    throw err;
  } finally {
    client.release();
  }
}

import type { ObjectStore } from "./storage.js";
import { FilesystemObjectStore } from "./storage.js";

export async function createObjectStoreFromEnv(): Promise<ObjectStore> {
  if (process.env.S3_BUCKET) {
    const { createS3ObjectStore } = await import("./s3.js");
    return createS3ObjectStore({
      bucket: process.env.S3_BUCKET,
      ...(process.env.S3_ENDPOINT ? { endpoint: process.env.S3_ENDPOINT } : {}),
      ...(process.env.S3_REGION ? { region: process.env.S3_REGION } : {}),
      ...(process.env.S3_ACCESS_KEY ? { accessKeyId: process.env.S3_ACCESS_KEY } : {}),
      ...(process.env.S3_SECRET_KEY ? { secretAccessKey: process.env.S3_SECRET_KEY } : {}),
      forcePathStyle: true,
      ...(process.env.CREATIVE_STORE_BASE_URL ? { publicBaseUrl: process.env.CREATIVE_STORE_BASE_URL } : {}),

  }
  return new FilesystemObjectStore(process.env.CREATIVE_STORE_ROOT ?? "./storage/creatives");
