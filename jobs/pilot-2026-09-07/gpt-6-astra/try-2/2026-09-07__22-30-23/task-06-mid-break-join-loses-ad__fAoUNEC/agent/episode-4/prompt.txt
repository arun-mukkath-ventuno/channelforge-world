// Avail detection over an origin media playlist (PRD §9, §22, §41).
//
// Walks the playlist, finds each CUE-OUT…CUE-IN region, and assigns a
// *deterministic* opportunity id derived from the channel + the media sequence
// of the avail's first segment. Determinism matters: every viewer's data-plane
// request and the decisioning path must agree on the same opportunity id for a
// given break without coordinating, so the cached pod is found on the hot path.

import { parseMarkerLine } from "@ssai/core";

export interface Avail {
  opportunityId: string;
  channelId: string;
  /** The producer's stable break id, from the CUE-OUT's accompanying DATERANGE ID (ChannelForge
   *  threads its break_id there). Undefined when the origin carried no DATERANGE ID. This is what
   *  lets ChannelForge reconcile a delivery back to the exact planned break (its opportunityId,
   *  by contrast, is a media-sequence-derived id independent of the producer). */
  externalBreakId?: string;
  /** Break duration: the marker's declared duration, or the summed segment time. */
  duration: number;
  /** Declared duration from the marker, if it carried a valid one. */
  declaredDuration?: number;
  /** Actual duration summed from the segments inside the avail. */
  measuredDuration: number;
  /** Media sequence number of the first segment inside the avail. */
  startSequence: number;
  programDateTime?: string;
  /** Line index of the CUE-OUT marker. */
  startLine: number;
  /** Line index of the CUE-IN marker. */
  endLine: number;
}

const MEDIA_SEQUENCE = /^#EXT-X-MEDIA-SEQUENCE:(\d+)/;
const EXTINF = /^#EXTINF:([\d.]+)/;
const PROGRAM_DATE_TIME = /^#EXT-X-PROGRAM-DATE-TIME:(.+)/;

/** Accepts `60`, `60.0`, `DURATION=60`, or `DURATION=60,...`. Returns null if absent/invalid. */
function parseCueOutDuration(value: string): number | null {
  if (value.length === 0) return null;
  if (SECONDS.test(value)) {
    const n = Number(value);
    return n >= 0 ? n : null;
  }
  // Attribute form: DURATION=60 (possibly with more comma-separated attrs).
  return parseAttr(value, "DURATION");
}

/** Parse `KEY=value` (optionally quoted) out of a comma-separated attribute list. */
function parseAttr(attrs: string, key: string): number | null {
  const re = new RegExp(`${key}=("?)(-?\\d+(?:\\.\\d+)?)\\1`, "i");
  const m = attrs.match(re);
  if (!m) return null;
  const n = Number(m[2]);
  return Number.isFinite(n) && n >= 0 ? n : null;
}

/** Parse a string `KEY="value"` (or bare `KEY=value`) attribute; null if absent/empty. */
function parseStringAttr(attrs: string, key: string): string | null {
  const re = new RegExp(`${key}=(?:"([^"]*)"|([^,\\s]+))`, "i");
  const m = attrs.match(re);
  if (!m) return null;
  const value = m[1] ?? m[2] ?? "";
  return value.length > 0 ? value : null;
}

// Manifest stitching (PRD §22-25). Given an origin media playlist, its detected
// avails, and the *resolved* pod media for each avail, replace each avail's
// CUE-OUT…CUE-IN region with the pod's segments (real creative segments for ads,
// synthesized for slate). Avails without resolved media are left as origin (their
// ChannelForge slate plays) — playback survives ad failure (§42).
//
// Pure and synchronous by design: no DB, no transcode, no network. Segment media
// is resolved beforehand (see media.ts) so this stays a lean string transform.

import type { Avail } from "./avails.js";
import type { RenderedPodItem } from "./media.js";

export interface StitchOptions {
  /** Emit EXT-X-DATERANGE ad metadata for observability (PRD §25). Default true. */
  emitAdMetadata?: boolean;
}

export interface StitchResult {
  manifest: string;
  /** Opportunity ids that were stitched (had resolved media). */
  stitched: string[];
}

/** Replace each avail that has resolved pod media; leave the rest untouched. */
export function stitchManifest(
  origin: string,
  avails: Avail[],
  rendered: Map<string, RenderedPodItem[]>,
  opts: StitchOptions = {},
): StitchResult {
  const emitMeta = opts.emitAdMetadata ?? true;
  const lines = origin.split("\n");

  const availByStart = new Map<number, Avail>();
  for (const a of avails) availByStart.set(a.startLine, a);

  const out: string[] = [];
  const stitched: string[] = [];

  for (let i = 0; i < lines.length; i++) {
    const avail = availByStart.get(i);
    const pod = avail ? rendered.get(avail.opportunityId) : undefined;

    if (avail && pod && pod.length > 0) {
      out.push(...renderPod(pod, emitMeta, avail.programDateTime));
      stitched.push(avail.opportunityId);
      i = avail.endLine; // skip the origin avail region (CUE-OUT…CUE-IN inclusive)
      continue;
    }
    out.push(lines[i]!);
  }

  return { manifest: out.join("\n"), stitched };
}

function renderPod(pod: RenderedPodItem[], emitMeta: boolean, breakStart?: string): string[] {
  const out: string[] = [];
  for (const item of pod) {
    out.push("#EXT-X-DISCONTINUITY"); // timing/encoding boundary (PRD §24)

    if (item.type === "ad" && emitMeta && item.creativeId) {
      const start = breakStart
        ? new Date(new Date(breakStart).getTime() + item.startOffset * 1000).toISOString()
        : undefined;
      out.push(
        `#EXT-X-DATERANGE:ID="${item.creativeId}",CLASS="com.fastworld.ad",` +
          (start ? `START-DATE="${start}",` : "") +
          `DURATION=${item.duration},X-AD-ID="${item.creativeId}"` +
          (item.campaignId ? `,X-CAMPAIGN-ID="${item.campaignId}"` : ""),
      );
    }

    for (const seg of item.segments) {
      out.push(`#EXTINF:${seg.duration.toFixed(3)},`);
      out.push(seg.uri);
    }
  }
  out.push("#EXT-X-DISCONTINUITY"); // rejoin origin content (PRD §24)
  return out;
}
