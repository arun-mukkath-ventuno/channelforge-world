import type { PodItem } from "@ssai/core";
import { Redis } from "ioredis";

export interface CachedPod {
  sessionId: string;
  opportunityId: string;
  channelId: string;
  breakDuration: number;
  pod: PodItem[];
}

export interface PodCache {
  get(sessionId: string, opportunityId: string): Promise<CachedPod | null>;
  set(entry: CachedPod, ttlSeconds?: number): Promise<void>;
  del(sessionId: string, opportunityId: string): Promise<void>;
}

function key(sessionId: string, opportunityId: string): string {
  return `pod:${sessionId}:${opportunityId}`;
}

/** Default pod TTL: long enough to outlast a break, short enough to self-clean. */
export const DEFAULT_POD_TTL_SECONDS = 900;

export class RedisPodCache implements PodCache {
  constructor(private readonly redis: Redis) {}

  async get(sessionId: string, opportunityId: string): Promise<CachedPod | null> {
    const raw = await this.redis.get(key(sessionId, opportunityId));
    return raw ? (JSON.parse(raw) as CachedPod) : null;
  }

  async set(entry: CachedPod, ttlSeconds = DEFAULT_POD_TTL_SECONDS): Promise<void> {
    await this.redis.set(key(entry.sessionId, entry.opportunityId), JSON.stringify(entry), "EX", ttlSeconds);
  }

  async del(sessionId: string, opportunityId: string): Promise<void> {
    await this.redis.del(key(sessionId, opportunityId));
  }
}

export class MemoryPodCache implements PodCache {
  private readonly store = new Map<string, CachedPod>();

  async get(sessionId: string, opportunityId: string): Promise<CachedPod | null> {
    return this.store.get(key(sessionId, opportunityId)) ?? null;
  }

  async set(entry: CachedPod): Promise<void> {
    this.store.set(key(entry.sessionId, entry.opportunityId), entry);
  }

  async del(sessionId: string, opportunityId: string): Promise<void> {
    this.store.delete(key(sessionId, opportunityId));
  }
}
