# Scope specificity for deterministic precedence ordering (higher = more specific).
_SPECIFICITY = {"asset": 4, "series": 3, "collection": 2, "channel": 1, "agreement": 0}


def create_policy(db: Session, **fields) -> RightsPolicy:
    policy = RightsPolicy(**fields)
    db.add(policy)
    db.flush()
    return policy


def list_policies(db: Session, org_id: uuid.UUID) -> list[RightsPolicy]:
    return list(
        db.scalars(
            select(RightsPolicy)
            .where(RightsPolicy.organization_id == org_id)
            .order_by(RightsPolicy.created_at.desc())
        )
    )


def add_version(db: Session, *, policy: RightsPolicy, **fields) -> RightsPolicyVersion:
    """Append the next immutable version to a policy."""
    current_max = db.scalar(
        select(func.max(RightsPolicyVersion.version_number)).where(
            RightsPolicyVersion.policy_id == policy.id
        )
    )
    version = RightsPolicyVersion(
        organization_id=policy.organization_id,
        policy_id=policy.id,
        version_number=(current_max or 0) + 1,
        **fields,
    )
    db.add(version)
    db.flush()
    return version


def assign(
    db: Session, *, org_id: uuid.UUID, policy_id: uuid.UUID, scope_type: str,
    scope_id: uuid.UUID, priority: int = 0,
) -> RightsAssignment:
    assignment = RightsAssignment(
        organization_id=org_id, policy_id=policy_id, scope_type=scope_type,
        scope_id=scope_id, priority=priority,
    )
    db.add(assignment)
    db.flush()
    return assignment


def _latest_version(db: Session, policy_id: uuid.UUID) -> RightsPolicyVersion | None:
    return db.scalar(
        select(RightsPolicyVersion)
        .where(RightsPolicyVersion.policy_id == policy_id)
        .order_by(RightsPolicyVersion.version_number.desc())
        .limit(1)
    )


def resolve_applicable(
    db: Session,
    *,
    org_id: uuid.UUID,
    asset: Asset,
    channel_id: uuid.UUID | None = None,
    agreement_id: uuid.UUID | None = None,
) -> list[ApplicablePolicy]:
    """Return the policies that apply to this asset in context, most-specific first.

    Matches assignments scoped to the asset, its series, the channel, or the agreement, using each
    policy's latest version. (Collection-scoped assignments are resolved in a later increment.)
    """
    scope_targets: list[tuple[str, uuid.UUID]] = [("asset", asset.id)]
    if asset.series_id is not None:
        scope_targets.append(("series", asset.series_id))
    if channel_id is not None:
        scope_targets.append(("channel", channel_id))
    if agreement_id is not None:
        scope_targets.append(("agreement", agreement_id))

    applicable: list[ApplicablePolicy] = []
    for scope_type, scope_id in scope_targets:
        assignments = db.scalars(
            select(RightsAssignment).where(
                RightsAssignment.organization_id == org_id,
                RightsAssignment.scope_type == scope_type,
                RightsAssignment.scope_id == scope_id,
                RightsAssignment.enabled.is_(True),
            )
        )
        for assignment in assignments:
            version = _latest_version(db, assignment.policy_id)
            if version is not None:
                applicable.append(
                    ApplicablePolicy(
                        version=version, scope_type=scope_type, priority=assignment.priority
                    )
                )

    applicable.sort(
        key=lambda a: (_SPECIFICITY.get(a.scope_type, 0), a.priority), reverse=True
    )
    return applicable

def get_db() -> Iterator[Session]:

    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
