import type { ObjectStore } from "@ssai/storage";
import { segmentDurations } from "./segment.js";

export interface RenderedSegment {
  duration: number;
  uri: string;
}

export interface RenderedPodItem {
  type: "ad" | "slate";
  creativeId?: string;
  campaignId?: string;
  duration: number;
  startOffset: number;
  segments: RenderedSegment[];
}

export interface ResolveOptions {
  /** Channel id, used to pick the channel slate (PRD §21). */
  channelId?: string;
  /** URL prefix the data plane serves stored creative segments from. */
  segmentBaseUrl?: string;
  /** URL prefix the data plane serves stored slate segments from. */
  slateBaseUrl?: string;
  /** Slate ids to try in order; defaults to [channelId, "default"]. */
  slateFallbacks?: string[];
  /** Fallback slate segment length when no slate media exists at all. */
  slateSegmentDuration?: number;
}

/** Extract EXTINF duration + segment filename pairs from a media playlist. */
export function parseVariantPlaylist(m3u8: string): RenderedSegment[] {
  const lines = m3u8.split("\n");
  const out: RenderedSegment[] = [];
  let pendingDuration: number | null = null;
  for (const raw of lines) {
    const line = raw.trim();
    const inf = line.match(/^#EXTINF:([\d.]+)/);
    if (inf) {
      pendingDuration = Number(inf[1]);
      continue;
    }
    if (line.length > 0 && !line.startsWith("#")) {
      out.push({ duration: pendingDuration ?? 0, uri: line });
      pendingDuration = null;
    }
  }
  return out;
}

export async function resolvePodSegments(
  pod: PodItem[],
  store: ObjectStore,
  opts: ResolveOptions = {},
): Promise<RenderedPodItem[]> {
  const base = (opts.segmentBaseUrl ?? "/v1/segments").replace(/\/$/, "");
  const slateBase = (opts.slateBaseUrl ?? "/v1/slate").replace(/\/$/, "");
  const slateSegDur = opts.slateSegmentDuration ?? 6;
  const slateIds = opts.slateFallbacks ?? [opts.channelId, "default"].filter((x): x is string => !!x);
  const rendered: RenderedPodItem[] = [];

  // Load the slate clip once per pod (reused across slate items).
  const slateClip = await loadSlateVariant(slateIds, store, slateBase);

  for (const item of pod) {
    if (item.type === "ad" && item.creativeId) {
      const segments = await loadCreativeSegments(item.creativeId, store, base);
      if (segments.length > 0) {
        rendered.push({
          type: "ad",
          creativeId: item.creativeId,
          ...(item.campaignId ? { campaignId: item.campaignId } : {}),
          duration: item.duration,
          startOffset: item.startOffset,
          segments,
        });
        continue;
      }
      // Creative not prepared/stored → degrade this slot to slate (PRD §42).
    }

    rendered.push({
      type: "slate",
      duration: item.duration,
      startOffset: item.startOffset,
      segments: slateClip
        ? fillWithSlate(slateClip, item.duration)
        : synthesizeSlate(item.duration, slateSegDur, slateBase),
    });
  }

  return rendered;
}

interface SlateClip {
  slateId: string;
  base: string;
  segments: RenderedSegment[]; // one loop of the slate, uris already absolute
}

async function loadSlateVariant(slateIds: string[], store: ObjectStore, base: string): Promise<SlateClip | null> {
  for (const slateId of slateIds) {
    try {
      const playlist = (await store.get(`slates/${slateId}/${slateId}.m3u8`)).toString("utf8");
      const segs = parseVariantPlaylist(playlist);
      if (segs.length > 0) {
        return {
          slateId,
          base,
          segments: segs.map((s) => ({ duration: s.duration, uri: `${base}/${slateId}/${s.uri}` })),
        };
      }
    } catch {
      // try the next fallback
    }
  }
  return null;
}

/** Loop the slate clip's segments to cover `duration` (exact when duration is a
 *  multiple of the clip length; otherwise overshoots by at most one segment). */
function fillWithSlate(clip: SlateClip, duration: number): RenderedSegment[] {
  const out: RenderedSegment[] = [];
  let total = 0;
  let i = 0;
  const guard = clip.segments.length * 10000;
  while (total < duration - 0.01 && i < guard) {
    const seg = clip.segments[i % clip.segments.length]!;
    out.push(seg);
    total += seg.duration;
    i++;
  }
  return out;
}

async function loadCreativeSegments(creativeId: string, store: ObjectStore, base: string): Promise<RenderedSegment[]> {
  try {
    const playlist = (await store.get(`creatives/${creativeId}/${creativeId}.m3u8`)).toString("utf8");
    return parseVariantPlaylist(playlist).map((s) => ({
      duration: s.duration,
      uri: `${base}/${creativeId}/${s.uri}`,
    }));
  } catch {
    return [];
  }
}

/** Slate has no prepared media yet; emit synthetic segment refs. */
function synthesizeSlate(duration: number, segDur: number, base: string): RenderedSegment[] {
  return segmentDurations(duration, segDur).map((d, i) => ({
    duration: d,
    uri: `${base}/_slate/slate_${String(i).padStart(3, "0")}.ts`,
  }));
}
