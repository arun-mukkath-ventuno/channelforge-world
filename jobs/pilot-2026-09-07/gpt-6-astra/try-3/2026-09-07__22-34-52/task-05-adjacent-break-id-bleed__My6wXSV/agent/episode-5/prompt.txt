  it("parses bare-number CUE-OUT: #EXT-X-CUE-OUT:60", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:60", 5);
    expect(m).toMatchObject({ kind: "cue_out", duration: 60, lineIndex: 5 });

  it("parses attribute-form CUE-OUT: #EXT-X-CUE-OUT:DURATION=60", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:DURATION=60", 0);
    expect(m).toMatchObject({ kind: "cue_out", duration: 60 });
  });

  it("parses DURATION with extra attributes", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:DURATION=90,SCTE35=abc", 0);
    expect(m?.duration).toBe(90);
  });

  it("parses fractional durations", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:29.97", 0);
    expect(m?.duration).toBeCloseTo(29.97);
  });

  it("parses CUE-IN with no attributes", () => {
    const m = parseMarkerLine("#EXT-X-CUE-IN", 9);
    expect(m).toMatchObject({ kind: "cue_in", lineIndex: 9 });
    expect(m?.duration).toBeUndefined();
  });

  it("ignores CUE-OUT-CONT as a new avail", () => {
