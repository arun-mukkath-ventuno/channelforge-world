import { describe, it, expect } from "vitest";
import { parseMarkerLine } from "./scte.js";

describe("parseMarkerLine — cue markers (PRD §9, §60 task 3)", () => {
  it("parses bare-number CUE-OUT: #EXT-X-CUE-OUT:60", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:60", 5);
    expect(m).toMatchObject({ kind: "cue_out", duration: 60, lineIndex: 5 });
  });

  it("parses attribute-form CUE-OUT: #EXT-X-CUE-OUT:DURATION=60", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:DURATION=60", 0);
    expect(m).toMatchObject({ kind: "cue_out", duration: 60 });
  });

  it("parses DURATION with extra attributes", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:DURATION=90,SCTE35=abc", 0);
    expect(m?.duration).toBe(90);
  });

  it("parses fractional durations", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:29.97", 0);
    expect(m?.duration).toBeCloseTo(29.97);
  });

  it("parses CUE-IN with no attributes", () => {
    const m = parseMarkerLine("#EXT-X-CUE-IN", 9);
    expect(m).toMatchObject({ kind: "cue_in", lineIndex: 9 });
    expect(m?.duration).toBeUndefined();
  });

  it("ignores CUE-OUT-CONT as a new avail", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT-CONT:ElapsedTime=30,Duration=60", 0);
    expect(m).toBeNull();
  });

  it("flags an unparseable CUE-OUT duration instead of throwing (PRD §41)", () => {
    const m = parseMarkerLine("#EXT-X-CUE-OUT:oops", 0);
    expect(m).toMatchObject({ kind: "cue_out", invalidDuration: true });
    expect(m?.duration).toBeUndefined();
  });

  it("returns null for non-marker lines", () => {
    expect(parseMarkerLine("#EXTINF:6.0,", 0)).toBeNull();
    expect(parseMarkerLine("segment100.ts", 0)).toBeNull();
  });

  it("parses DATERANGE with a DURATION attribute", () => {
    const line = '#EXT-X-DATERANGE:ID="ad1",DURATION=30,SCTE35-OUT=0xFC';
    const m = parseMarkerLine(line, 3);
    expect(m).toMatchObject({ kind: "daterange", duration: 30 });
  });

  it("extracts the DATERANGE ID (ChannelForge threads its break_id there)", () => {
    const line = '#EXT-X-DATERANGE:ID="BRK-a11e5310-0003",PLANNED-DURATION=90,SCTE35-OUT=0xFC30';
    const m = parseMarkerLine(line, 7);
    expect(m).toMatchObject({ kind: "daterange", id: "BRK-a11e5310-0003", duration: 90 });
  });

  it("omits id when the DATERANGE carries none", () => {
    const m = parseMarkerLine("#EXT-X-DATERANGE:DURATION=30", 0);
    expect(m?.kind).toBe("daterange");
    expect(m?.id).toBeUndefined();
  });
});
