import { describe, it, expect } from "vitest";
import { stitchManifest } from "./stitch.js";
import { detectAvails } from "./avails.js";
import type { RenderedPodItem } from "./media.js";

const ORIGIN = [
  "#EXTM3U",
  "#EXT-X-VERSION:3",
  "#EXT-X-TARGETDURATION:6",
  "#EXT-X-MEDIA-SEQUENCE:100",
  "#EXTINF:6.000,",
  "segment100.ts",
  "#EXTINF:6.000,",
  "segment101.ts",
  "#EXT-X-CUE-OUT:30",
  "#EXTINF:6.000,",
  "slate001.ts",
  "#EXTINF:6.000,",
  "slate002.ts",
  "#EXT-X-CUE-IN",
  "#EXTINF:6.000,",
  "segment105.ts",
  "",
].join("\n");

const AD_MEDIA: RenderedPodItem[] = [
  {
    type: "ad",
    creativeId: "cr_a",
    campaignId: "camp_a",
    duration: 30,
    startOffset: 0,
    segments: [
      { duration: 8.333, uri: "/v1/segments/cr_a/cr_a_000.ts" },
      { duration: 8.333, uri: "/v1/segments/cr_a/cr_a_001.ts" },
    ],
  },
];

function renderedFor(oppId: string, items: RenderedPodItem[]): Map<string, RenderedPodItem[]> {
  return new Map([[oppId, items]]);
}

describe("stitchManifest (PRD §22-25)", () => {
  it("replaces the avail region with real creative segment URLs", () => {
    const avails = detectAvails(ORIGIN, "channel_food");
    const { manifest, stitched } = stitchManifest(ORIGIN, avails, renderedFor(avails[0]!.opportunityId, AD_MEDIA));
    expect(stitched).toEqual([avails[0]!.opportunityId]);
    expect(manifest).not.toContain("slate001.ts");
    expect(manifest).not.toContain("#EXT-X-CUE-OUT");
    expect(manifest).toContain("/v1/segments/cr_a/cr_a_000.ts");
    expect(manifest).toContain("#EXTINF:8.333,");
    // Surrounding content preserved.
    expect(manifest).toContain("segment101.ts");
    expect(manifest).toContain("segment105.ts");
  });

  it("emits discontinuities and ad metadata (PRD §24, §25)", () => {
    const avails = detectAvails(ORIGIN, "channel_food");
    const { manifest } = stitchManifest(ORIGIN, avails, renderedFor(avails[0]!.opportunityId, AD_MEDIA));
    expect(manifest).toContain("#EXT-X-DISCONTINUITY");
    expect(manifest).toContain('X-AD-ID="cr_a"');
    expect(manifest).toContain('X-CAMPAIGN-ID="camp_a"');
  });

  it("leaves an avail untouched when no media is resolved (origin slate plays, §42)", () => {
    const avails = detectAvails(ORIGIN, "channel_food");
    const { manifest, stitched } = stitchManifest(ORIGIN, avails, new Map());
    expect(stitched).toEqual([]);
    expect(manifest).toContain("slate001.ts");
    expect(manifest).toBe(ORIGIN);
  });
});
