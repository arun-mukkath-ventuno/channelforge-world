// Same-origin reverse proxy for ChannelForge HLS playout.
//
// ChannelForge's HLS origin (forge.ventunotech.com) currently serves no CORS
// headers, so a browser MSE player (Shaka on Chrome/Firefox) can't fetch its
// segments cross-origin. Proxying the manifest + segments through this route
// makes them same-origin, adds CORS, and fixes the segment MIME type. Because
// the manifest uses relative segment names (index1103.ts), mirroring the path
// under /api/fast/hls/* means those relative refs resolve back through here
// with no manifest rewriting needed.
//
// Production-grade path is to enable CORS (+ correct MIME) on ChannelForge and
// point the player straight at the origin; this proxy is the zero-touch bridge.

const FORGE_BASE = process.env.FORGE_HLS_BASE || "https://forge.ventunotech.com/hls";

export const dynamic = "force-dynamic";

function contentType(path: string): string | null {
  if (path.endsWith(".m3u8")) return "application/vnd.apple.mpegurl";
  if (path.endsWith(".ts")) return "video/mp2t";
  if (path.endsWith(".m4s") || path.endsWith(".mp4")) return "video/mp4";
  if (path.endsWith(".aac")) return "audio/aac";
  return null;
}

export async function GET(
  req: Request,
  { params }: { params: Promise<{ path: string[] }> }
) {
  const { path } = await params;
  const rel = (path || []).join("/");
  // Guard against path traversal / absolute-URL injection.
  if (!rel || rel.includes("..") || rel.includes("://")) {
    return new Response("bad path", { status: 400 });
  }

  const upstream = `${FORGE_BASE}/${rel}`;
  const range = req.headers.get("range");

  let res: Response;
  try {
    res = await fetch(upstream, {
      cache: "no-store",
      headers: range ? { range } : undefined,
    });
  } catch {
    return new Response("upstream fetch failed", { status: 502 });
  }

  const headers = new Headers();
  const ct = contentType(rel) || res.headers.get("content-type") || "application/octet-stream";
  headers.set("content-type", ct);
  headers.set("access-control-allow-origin", "*");
  headers.set("cache-control", "no-store");
  const cr = res.headers.get("content-range");
  if (cr) headers.set("content-range", cr);
  headers.set("accept-ranges", res.headers.get("accept-ranges") || "bytes");

  return new Response(res.body, { status: res.status, headers });
}

export async function OPTIONS() {
  return new Response(null, {
    status: 204,
    headers: {
      "access-control-allow-origin": "*",
      "access-control-allow-methods": "GET, OPTIONS",
      "access-control-allow-headers": "range",
    },
  });
}
