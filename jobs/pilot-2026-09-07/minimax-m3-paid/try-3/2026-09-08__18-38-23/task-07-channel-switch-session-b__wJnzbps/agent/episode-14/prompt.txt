-- FAST World SSAI — initial schema (PRD §44-47).
-- Postgres. Append-only where noted; decision/pod payloads retained in full for
-- debugging and RL evaluation.

CREATE TABLE channels (
  id                TEXT PRIMARY KEY,
  name              TEXT NOT NULL,
  origin_url        TEXT NOT NULL,
  slate_id          TEXT,
  status            TEXT NOT NULL DEFAULT 'active',
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE slates (
  id                TEXT PRIMARY KEY,
  scope             TEXT NOT NULL,              -- 'channel' | 'network' | 'emergency'
  channel_id        TEXT REFERENCES channels(id),
  hls_path          TEXT NOT NULL,              -- prepared HLS media (same profile as channel)
  duration          NUMERIC NOT NULL,
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE campaigns (
  id                TEXT PRIMARY KEY,
  advertiser        TEXT NOT NULL,
  start_date        TIMESTAMPTZ NOT NULL,
  end_date          TIMESTAMPTZ NOT NULL,
  status            TEXT NOT NULL DEFAULT 'active',   -- 'active' | 'paused' | 'archived'
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE campaign_channels (
  campaign_id       TEXT NOT NULL REFERENCES campaigns(id) ON DELETE CASCADE,
  channel_id        TEXT NOT NULL REFERENCES channels(id) ON DELETE CASCADE,
  PRIMARY KEY (campaign_id, channel_id)
);

CREATE TABLE creatives (
  id                TEXT PRIMARY KEY,
  campaign_id       TEXT NOT NULL REFERENCES campaigns(id) ON DELETE CASCADE,
  name              TEXT NOT NULL,
  duration          NUMERIC NOT NULL,
  status            TEXT NOT NULL DEFAULT 'uploaded',  -- uploaded|validating|transcoding|ready|failed
  priority          INTEGER NOT NULL DEFAULT 0,
  weight            INTEGER NOT NULL DEFAULT 100,
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Pre-transcoded HLS variants (PRD §14). SSAI never transcodes on the hot path.
CREATE TABLE creative_variants (
  id                TEXT PRIMARY KEY,
  creative_id       TEXT NOT NULL REFERENCES creatives(id) ON DELETE CASCADE,
  bandwidth         INTEGER,
  resolution        TEXT,
  hls_path          TEXT NOT NULL,
  segment_duration  NUMERIC
);

CREATE TABLE playback_sessions (
  id                TEXT PRIMARY KEY,
  channel_id        TEXT NOT NULL REFERENCES channels(id),
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
  last_activity_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  device_type       TEXT,
  country           TEXT,
  user_id           TEXT,
  anonymous_id      TEXT,
  ip_hash           TEXT,
  user_agent        TEXT
);

-- PRD §45.
CREATE TABLE ad_opportunities (
  id                TEXT PRIMARY KEY,
  channel_id        TEXT NOT NULL REFERENCES channels(id),
  program_id        TEXT,
  start_time        TIMESTAMPTZ NOT NULL,
  duration          NUMERIC NOT NULL,
  marker_type       TEXT NOT NULL,             -- 'cue_out' | 'daterange' | 'scte35'
  marker_payload    TEXT,
  detected_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  status            TEXT NOT NULL DEFAULT 'detected'
);

-- PRD §46. Full decision payload retained for debugging + RL evaluation.
CREATE TABLE ad_decisions (
  id                TEXT PRIMARY KEY,
  opportunity_id    TEXT NOT NULL REFERENCES ad_opportunities(id),
  session_id        TEXT NOT NULL REFERENCES playback_sessions(id),
  requested_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  completed_at      TIMESTAMPTZ,
  eligible_count    INTEGER NOT NULL DEFAULT 0,
  selected_count    INTEGER NOT NULL DEFAULT 0,
  seed              TEXT,
  decision_payload  JSONB NOT NULL
);

CREATE TABLE ad_pods (
  id                TEXT PRIMARY KEY,
  decision_id       TEXT NOT NULL REFERENCES ad_decisions(id) ON DELETE CASCADE,
  break_duration    NUMERIC NOT NULL,
  ad_seconds        NUMERIC NOT NULL,
  slate_seconds     NUMERIC NOT NULL
);

-- PRD §47.
CREATE TABLE ad_pod_items (
  id                BIGSERIAL PRIMARY KEY,
  pod_id            TEXT NOT NULL REFERENCES ad_pods(id) ON DELETE CASCADE,
  position          INTEGER NOT NULL,
  item_type         TEXT NOT NULL,             -- 'AD' | 'SLATE'
  creative_id       TEXT REFERENCES creatives(id),
  duration          NUMERIC NOT NULL,
  start_offset      NUMERIC NOT NULL,
  end_offset        NUMERIC NOT NULL
);

CREATE TABLE manifest_requests (
  id                BIGSERIAL PRIMARY KEY,

  channel_id        TEXT NOT NULL REFERENCES channels(id),
  requested_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  variant           TEXT,
  stitched          BOOLEAN NOT NULL DEFAULT false,
  generation_ms     NUMERIC
);
