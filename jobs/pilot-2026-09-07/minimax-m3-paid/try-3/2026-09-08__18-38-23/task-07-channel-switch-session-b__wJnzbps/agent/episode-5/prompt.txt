// Analytics (PRD §31, §32). Reads the append-only event stream, decisions/pods
// and opportunities to report streaming, opportunity, advertising and fill
// metrics. Queries avoid Postgres-only constructs (FILTER, EXTRACT EPOCH,
// interval literals) so the same code runs under pg-mem in tests; time math is
// done in JS.

import type { Queryable } from "./db.js";
import { num } from "./map.js";

export interface AnalyticsSummary {
  streaming: {
    totalSessions: number;
    activeSessions: number;
    sessionsByChannel: { channelId: string; sessions: number }[];
    avgSessionSeconds: number;
    viewingHours: number;
  };
  opportunities: {
    totalOpportunities: number;
    totalOpportunitySeconds: number;
    totalBreaks: number;
    filledBreaks: number;
    partiallyFilledBreaks: number;
    unfilledBreaks: number;
  };
  advertising: {
    impressions: number;
    starts: number;
    firstQuartile: number;
    midpoint: number;
    thirdQuartile: number;
    completes: number;
    errors: number;
    completionRate: number;
  };
  fill: {
    adSeconds: number;
    slateSeconds: number;
    durationFillRate: number;
    slateRate: number;
    opportunityFillRate: number;
  };
}

export interface AnalyticsOptions {
  /** A session is "active" if it had activity within this window. Default 300s. */
  activeWindowSeconds?: number;
  now?: Date;
}

function ratio(numerator: number, denominator: number): number {
  return denominator > 0 ? numerator / denominator : 0;
}

export async function getAnalytics(db: Queryable, opts: AnalyticsOptions = {}): Promise<AnalyticsSummary> {
  const now = opts.now ?? new Date();
  const activeWindow = opts.activeWindowSeconds ?? 300;
  const activeThreshold = new Date(now.getTime() - activeWindow * 1000).toISOString();

  // ── Streaming ────────────────────────────────────────────────────────────
  const sessionRows = (
    await db.query<{ channel_id: string; created_at: unknown; last_activity_at: unknown }>(
      "SELECT channel_id, created_at, last_activity_at FROM playback_sessions",
    )
  ).rows;

  const byChannel = new Map<string, number>();
  let activeSessions = 0;
  let totalLengthMs = 0;
  const thresholdMs = Date.parse(activeThreshold);
  for (const s of sessionRows) {
    byChannel.set(s.channel_id, (byChannel.get(s.channel_id) ?? 0) + 1);
    const created = new Date(s.created_at as string | Date).getTime();
    const last = new Date(s.last_activity_at as string | Date).getTime();
    totalLengthMs += Math.max(0, last - created);
    if (last >= thresholdMs) activeSessions += 1;
  }
  const totalSessions = sessionRows.length;
  const avgSessionSeconds = totalSessions > 0 ? totalLengthMs / totalSessions / 1000 : 0;

  // ── Opportunities ─────────────────────────────────────────────────────────
  const oppRow = (
    await db.query<{ c: number; s: number }>(
      "SELECT COUNT(*) AS c, COALESCE(SUM(duration), 0) AS s FROM ad_opportunities",
    )
  ).rows[0]!;

  // Delivered breaks (per-session pods) drive fill classification.
  const podRow = (
    await db.query<Record<string, number>>(
      `SELECT
         COUNT(*) AS breaks,
         COALESCE(SUM(ad_seconds), 0) AS ad_seconds,
         COALESCE(SUM(slate_seconds), 0) AS slate_seconds,
         SUM(CASE WHEN ad_seconds > 0 AND slate_seconds = 0 THEN 1 ELSE 0 END) AS filled,
         SUM(CASE WHEN ad_seconds > 0 AND slate_seconds > 0 THEN 1 ELSE 0 END) AS partial,
         SUM(CASE WHEN ad_seconds = 0 THEN 1 ELSE 0 END) AS unfilled
       FROM ad_pods`,
    )
  ).rows[0]!;

  const totalBreaks = num(podRow.breaks);
  const adSeconds = num(podRow.ad_seconds);
  const slateSeconds = num(podRow.slate_seconds);
  const filledBreaks = num(podRow.filled ?? 0);
  const partialBreaks = num(podRow.partial ?? 0);
  const unfilledBreaks = num(podRow.unfilled ?? 0);

  // ── Advertising (event stream) ────────────────────────────────────────────
  const eventRows = (
    await db.query<{ event_type: string; c: number }>(
      // 'ad_%' is a prefilter only; exact types are looked up below, so server
      // events like ad_selected that also match are harmlessly ignored.
      "SELECT event_type, COUNT(*) AS c FROM events WHERE event_type LIKE 'ad_%' GROUP BY event_type",
    )
  ).rows;
  const counts = new Map<string, number>();
  for (const r of eventRows) counts.set(r.event_type, num(r.c));
  const get = (t: string) => counts.get(t) ?? 0;

  const starts = get("ad_start");
  const completes = get("ad_complete");

  return {
    streaming: {
      totalSessions,
      activeSessions,
      sessionsByChannel: [...byChannel.entries()]
        .map(([channelId, sessions]) => ({ channelId, sessions }))
        .sort((a, b) => a.channelId.localeCompare(b.channelId)),
      avgSessionSeconds,
      viewingHours: totalLengthMs / 3_600_000,
    },
    opportunities: {
      totalOpportunities: num(oppRow.c),
      totalOpportunitySeconds: num(oppRow.s),
      totalBreaks,
      filledBreaks,
      partiallyFilledBreaks: partialBreaks,
      unfilledBreaks,
    },
    advertising: {
      impressions: get("ad_impression"),
      starts,
      firstQuartile: get("ad_first_quartile"),
      midpoint: get("ad_midpoint"),
      thirdQuartile: get("ad_third_quartile"),
      completes,
      errors: get("ad_error"),
      completionRate: ratio(completes, starts),
    },
    fill: {
      adSeconds,
      slateSeconds,
      durationFillRate: ratio(adSeconds, adSeconds + slateSeconds),
      slateRate: ratio(slateSeconds, adSeconds + slateSeconds),
      opportunityFillRate: ratio(filledBreaks + partialBreaks, totalBreaks),
    },
  };
}
