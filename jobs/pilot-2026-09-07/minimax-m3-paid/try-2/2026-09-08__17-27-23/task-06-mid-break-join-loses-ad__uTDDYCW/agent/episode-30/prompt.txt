     2	//
     3	// Walks the playlist, finds each CUE-OUT…CUE-IN region, and assigns a
     4	// *deterministic* opportunity id derived from the channel + the media sequence
     5	// of the avail's first segment. Determinism matters: every viewer's data-plane
     6	// request and the decisioning path must agree on the same opportunity id for a
     7	// given break without coordinating, so the cached pod is found on the hot path.

     9	import { parseMarkerLine } from "@ssai/core";

    11	export interface Avail {
    12	  opportunityId: string;
    13	  channelId: string;
    14	  /** The producer's stable break id, from the CUE-OUT's accompanying DATERANGE ID (ChannelForge
    15	   *  threads its break_id there). Undefined when the origin carried no DATERANGE ID. This is what
    16	   *  lets ChannelForge reconcile a delivery back to the exact planned break (its opportunityId,
    17	   *  by contrast, is a media-sequence-derived id independent of the producer). */
    18	  externalBreakId?: string;
    19	  /** Break duration: the marker's declared duration, or the summed segment time. */
    20	  duration: number;
    21	  /** Declared duration from the marker, if it carried a valid one. */
    22	  declaredDuration?: number;
    23	  /** Actual duration summed from the segments inside the avail. */
    24	  measuredDuration: number;
    25	  /** Media sequence number of the first segment inside the avail. */
    26	  startSequence: number;
    27	  programDateTime?: string;
    28	  /** Line index of the CUE-OUT marker. */
    29	  startLine: number;
    30	  /** Line index of the CUE-IN marker. */
    31	  endLine: number;
    32	}

    34	const MEDIA_SEQUENCE = /^#EXT-X-MEDIA-SEQUENCE:(\d+)/;
    35	const EXTINF = /^#EXTINF:([\d.]+)/;
    36	const PROGRAM_DATE_TIME = /^#EXT-X-PROGRAM-DATE-TIME:(.+)/;

    38	/** Detect all complete avails in the manifest. Incomplete trailing avails (a
    39	 *  CUE-OUT with no CUE-IN yet in the window) are ignored — PRD §41. */
    40	export function detectAvails(manifest: string, channelId: string): Avail[] {

    44	  let lastPdt: string | undefined;

    47	  // in the same block (CUE-OUT first), so this pairs them whichever order a producer uses.

    54	    startSequence: number;

    56	    measuredDuration: number;
    57	    pdt?: string;
    58	    externalBreakId?: string;

    60	
    61	  for (let i = 0; i < lines.length; i++) {
    62	    const line = lines[i]!;
    63	    const trimmed = line.trim();
    64	
    65	    const seqMatch = trimmed.match(MEDIA_SEQUENCE);
    66	    if (seqMatch) {
    67	      mediaSequence = Number(seqMatch[1]);
    68	      continue;
    69	    }
    70	
    71	    const pdtMatch = trimmed.match(PROGRAM_DATE_TIME);
    72	    if (pdtMatch) {
    73	      lastPdt = pdtMatch[1]!.trim();
    74	      // continue — PDT applies to the next segment.
    75	    }
    76	
    77	    const marker = parseMarkerLine(line, i);
    78	    if (marker?.kind === "daterange" && marker.id) {
    79	      blockBreakId = marker.id;

    81	      if (open && open.externalBreakId === undefined) open.externalBreakId = marker.id;
    82	      continue;
    83	    }

    87	        startSequence: mediaSequence,

    90	        ...(lastPdt === undefined ? {} : { pdt: lastPdt }),
    91	        // Covers a producer that emits DATERANGE before CUE-OUT within the same block.
    92	        ...(blockBreakId === undefined ? {} : { externalBreakId: blockBreakId }),
    93	      };
    94	      continue;
    95	    }

    97	      const declared = open.declaredDuration;

   100	        channelId,
   101	        ...(open.externalBreakId === undefined ? {} : { externalBreakId: open.externalBreakId }),
   102	        duration: declared ?? open.measuredDuration,
   103	        ...(declared === undefined ? {} : { declaredDuration: declared }),
   104	        measuredDuration: open.measuredDuration,
   105	        startSequence: open.startSequence,
   106	        ...(open.pdt === undefined ? {} : { programDateTime: open.pdt }),
   107	        startLine: open.startLine,
   108	        endLine: i,

   122	      mediaSequence += 1;
   123	      blockBreakId = undefined;
   124	    }
   125	  }
   126	
   127	  return avails;
