// Ad decision: eligibility (PRD §16) + pod construction (PRD §18-21) into a
// single reconstructable AdDecision (PRD §35, §46). Pure and deterministic.

import type { AdDecision, AdOpportunity, Campaign, Creative, SessionContext } from "./types.js";
import { evaluateEligibility } from "./eligibility.js";
import { buildPod } from "./pod.js";

export interface DecideInput {
  opportunity: AdOpportunity;
  session: SessionContext;
  creatives: Creative[];
  campaigns: Map<string, Campaign>;
  now: string;
  seed?: string;
}

export function decide(input: DecideInput): AdDecision {
  const { opportunity, session } = input;

  const { eligible, rejected } = evaluateEligibility({
    channelId: opportunity.channelId,
    breakDuration: opportunity.breakDuration,
    now: input.now,
    creatives: input.creatives,
    campaigns: input.campaigns,
  });

  const pod = buildPod({
    breakDuration: opportunity.breakDuration,
    eligible,
    ...(input.seed === undefined ? {} : { seed: input.seed }),
  });

  return {
    opportunityId: opportunity.opportunityId,
    sessionId: session.sessionId,
    channelId: opportunity.channelId,
    breakDuration: opportunity.breakDuration,
    eligible,
    rejected: [...rejected, ...pod.rejected],
    pod: pod.pod,
    adSeconds: pod.adSeconds,
    slateSeconds: pod.slateSeconds,
    ...(input.seed === undefined ? {} : { seed: input.seed }),
  };
}
