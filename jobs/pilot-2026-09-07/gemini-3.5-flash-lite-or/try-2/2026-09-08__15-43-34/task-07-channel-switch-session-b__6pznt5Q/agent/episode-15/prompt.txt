function freshDb(): Queryable {
  const mem = newDb();
  mem.public.none(SCHEMA); // load full schema (multi-statement)
  const { Pool } = mem.adapters.createPg();
  return new Pool() as unknown as Queryable;
}

describe("seed + inventory repositories", () => {

    expect(food?.targetChannels).toEqual(["channel_food"]);

describe("decision persistence round-trip (PRD §46, §47)", () => {
  it("persists a decided pod and reads the full payload back", async () => {
    await seedDatabase(db);
    const creatives = await listCreatives(db);
    const campaigns = await campaignsAsMap(db);

    const opportunity: AdOpportunity = {

    await ensureSession(db, { id: "sess_1", channelId: "channel_food", deviceType: "web", country: "IN" });
    await upsertOpportunity(db, opportunity);
    await saveDecision(db, {
      decisionId: "dec_1",

      decision,
      requestedAt: "2026-08-29T12:15:00Z",

    });

    const stored = await getDecisionsBySession(db, "sess_1");
