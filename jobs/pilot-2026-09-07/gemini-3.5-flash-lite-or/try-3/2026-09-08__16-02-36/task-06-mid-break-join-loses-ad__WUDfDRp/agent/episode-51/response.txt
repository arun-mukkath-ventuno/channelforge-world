```ts
export function detectAvails(manifest: string, channelId: string): Avail[] {
  const lines = manifest.split("\n");

  let mediaSequence = 0;
  let lastPdt: string | undefined;
  let blockBreakId: string | undefined;
  const avails: Avail[] = [];

  let open: {
    startLine: number;
    startSequence: number;
    declaredDuration?: number;
    measuredDuration: number;
    pdt?: string;
    externalBreakId?: string;
  } | null = null;

  let windowStartSequence = 0;
  let midBreakDuration = 0;
  let midBreakExternalId: string | undefined;
  let midBreakStartLine = 0;
  let midBreakStartSequence = 0;

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i]!;
    const trimmed = line.trim();

    const seqMatch = trimmed.match(MEDIA_SEQUENCE);
    if (seqMatch) {
      mediaSequence = Number(seqMatch[1]);
      if (!open && midBreakDuration === 0) {
        windowStartSequence = mediaSequence;
        midBreakStartSequence = mediaSequence;
        midBreakStartLine = i;
      }
      continue;
    }

    const pdtMatch = trimmed.match(PROGRAM_DATE_TIME);
    if (pdtMatch) {
      lastPdt = pdtMatch[1]!.trim();
      continue;
    }

    const marker = parseMarkerLine(line, i);
    if (marker?.kind === "daterange" && marker.id) {
      blockBreakId = marker.id;
      if (open && open.externalBreakId === undefined) open.externalBreakId = marker.id;
      if (!open && midBreakExternalId === undefined) midBreakExternalId = marker.id;
      continue;
    }
    if (marker?.kind === "cue_out" && !open) {
      open = {
        startLine: i,
        startSequence: mediaSequence,
        measuredDuration: 0,
        ...(marker.duration === undefined ? {} : { declaredDuration: marker.duration }),
        ...(lastPdt === undefined ? {} : { pdt: lastPdt }),
        ...(blockBreakId === undefined ? {} : { externalBreakId: blockBreakId }),
      };
      // Reset midBreak state if a proper CUE-OUT is found
      midBreakDuration = 0;
      continue;
    }
    if (marker?.kind === "cue_in") {
      if (open) {
        const declared = open.declaredDuration;
        const extId = open.externalBreakId;
        const startSeq = open.startSequence;
        avails.push({
          opportunityId: extId ? `opp_${channelId}_${extId}` : `opp_${channelId}_${startSeq}`,
          channelId,
          ...(extId === undefined ? {} : { externalBreakId: extId }),
          duration: declared ?? open.measuredDuration,
          ...(declared === undefined ? {} : { declaredDuration: declared }),
          measuredDuration: open.measuredDuration,
          startSequence: startSeq,
          ...(open.pdt === undefined ? {} : { programDateTime: open.pdt }),
          startLine: open.startLine,
          endLine: i,
        });
        open = null;
      } else if (midBreakDuration > 0) {
        // Mid-break join: avail started before manifest window, ends with CUE-IN
        const extId = midBreakExternalId;
        const startSeq = midBreakStartSequence;
        avails.push({
          opportunityId: extId ? `opp_${channelId}_${extId}` : `opp_${channelId}_${startSeq}`,
          channelId,
          ...(extId === undefined ? {} : { externalBreakId: extId }),
          duration: midBreakDuration,
          measuredDuration: midBreakDuration,
          startSequence: startSeq,
          startLine: midBreakStartLine,
          endLine: i,
        });
        midBreakDuration = 0;
        midBreakExternalId = undefined;
      }
      continue;
    }

    const infMatch = trimmed.match(EXTINF);
    if (infMatch) {
      const segDur = Number(infMatch[1]);
      if (open) {
        open.measuredDuration += segDur;
      } else {
        if (midBreakDuration === 0) {
          midBreakStartSequence = mediaSequence;
          midBreakStartLine = i;
        }
        midBreakDuration += segDur;
      }
    } else if (trimmed.length > 0 && !trimmed.startsWith("#")) {
      mediaSequence += 1;
      blockBreakId = undefined;
    }
  }

  return avails;
}
