// Server-side ad tracking beacon queue (VAST). Durable + deduped delivery of third-party tracking
// URLs the server fires on behalf of the player.

import type { Queryable } from "../db.js";
import { num } from "../map.js";

export interface BeaconInput {
  sessionId: string;
  opportunityId: string;
  creativeId: string | null;
  eventType: string;
  url: string;
}

export interface BeaconRow {
  id: number;
  sessionId: string;
  opportunityId: string;
  creativeId: string | null;
  eventType: string;
  url: string;
  status: string;
  attempts: number;
}

interface Row {
  id: number;
  session_id: string;
  opportunity_id: string;
  creative_id: string | null;
  event_type: string;
  url: string;
  status: string;
  attempts: number;
}

const toRow = (r: Row): BeaconRow => ({
  id: num(r.id),
  sessionId: r.session_id,
  opportunityId: r.opportunity_id,
  creativeId: r.creative_id,
  eventType: r.event_type,
  url: r.url,
  status: r.status,
  attempts: num(r.attempts),
});

/** Enqueue a beacon; the unique key dedups repeats (same session/opportunity/event/url). */
export async function enqueueBeacon(db: Queryable, b: BeaconInput): Promise<void> {
  await db.query(
    `INSERT INTO beacon_deliveries (session_id, opportunity_id, creative_id, event_type, url)
     VALUES ($1, $2, $3, $4, $5)
     ON CONFLICT (session_id, opportunity_id, event_type, url) DO NOTHING`,
    [b.sessionId, b.opportunityId, b.creativeId ?? null, b.eventType, b.url],
  );
}

export async function listPendingBeacons(db: Queryable, limit = 100): Promise<BeaconRow[]> {
  const rows = (
    await db.query<Row>(
      "SELECT * FROM beacon_deliveries WHERE status = 'pending' ORDER BY created_at LIMIT $1",
      [limit],
    )
  ).rows;
  return rows.map(toRow);
}

export async function markBeaconSent(db: Queryable, id: number): Promise<void> {
  await db.query(
    "UPDATE beacon_deliveries SET status = 'sent', attempts = attempts + 1, sent_at = now() WHERE id = $1",
    [id],
  );
}

/** Record a failed attempt; the row goes to 'failed' once it reaches maxAttempts, else stays
 *  'pending' for retry. Returns true when it became terminally 'failed'. */
export async function markBeaconFailed(
  db: Queryable, id: number, error: string, maxAttempts: number,
): Promise<boolean> {
  const rows = (
    await db.query<{ status: string }>(
      `UPDATE beacon_deliveries
       SET attempts = attempts + 1, last_error = $2,
           status = CASE WHEN attempts + 1 >= $3 THEN 'failed' ELSE 'pending' END
       WHERE id = $1
       RETURNING status`,
      [id, error, maxAttempts],
    )
  ).rows;
  return rows[0]?.status === "failed";
}

export async function beaconStats(db: Queryable): Promise<Record<string, number>> {
  const rows = (
    await db.query<{ status: string; c: number }>(
      "SELECT status, COUNT(*) AS c FROM beacon_deliveries GROUP BY status",
    )
  ).rows;
  const out: Record<string, number> = { pending: 0, sent: 0, failed: 0 };
  for (const r of rows) out[r.status] = num(r.c);
  return out;
}
