// of the avail's first segment. Determinism matters: every viewer's data-plane
// request and the decisioning path must agree on the same opportunity id for a
// given break without coordinating, so the cached pod is found on the hot path.

import { parseMarkerLine } from "@ssai/core";

export interface Avail {
  opportunityId: string;
  channelId: string;
  /** The producer's stable break id, from the CUE-OUT's accompanying DATERANGE ID (ChannelForge
   *  threads its break_id there). Undefined when the origin carried no DATERANGE ID. This is what
   *  lets ChannelForge reconcile a delivery back to the exact planned break (its opportunityId,
   *  by contrast, is a media-sequence-derived id independent of the producer). */
  externalBreakId?: string;
  /** Break duration: the marker's declared duration, or the summed segment time. */
  duration: number;
  /** Declared duration from the marker, if it carried a valid one. */
  declaredDuration?: number;
  /** Actual duration summed from the segments inside the avail. */
  measuredDuration: number;
  /** Media sequence number of the first segment inside the avail. */
  startSequence: number;
  programDateTime?: string;
  /** Line index of the CUE-OUT marker. */
  startLine: number;
  /** Line index of the CUE-IN marker. */
  endLine: number;
}

const MEDIA_SEQUENCE = /^#EXT-X-MEDIA-SEQUENCE:(\d+)/;
const EXTINF = /^#EXTINF:([\d.]+)/;
const PROGRAM_DATE_TIME = /^#EXT-X-PROGRAM-DATE-TIME:(.+)/;

/** Detect all complete avails in the manifest. Incomplete trailing avails (a
 *  CUE-OUT with no CUE-IN yet in the window) are ignored — PRD §41. */
export function detectAvails(manifest: string, channelId: string): Avail[] {
  const lines = manifest.split("\n");

  let mediaSequence = 0;
  let lastPdt: string | undefined;
  // The DATERANGE ID seen in the current segment's tag block, cleared at each segment boundary so
  // one break's id never leaks onto the next. ChannelForge emits the CUE-OUT and its DATERANGE-OUT
  // in the same block (CUE-OUT first), so this pairs them whichever order a producer uses.
  let blockBreakId: string | undefined;
  const avails: Avail[] = [];

  // Pending avail state while we walk from CUE-OUT to CUE-IN.
  let open: {
    startLine: number;
    startSequence: number;
    declaredDuration?: number;
    measuredDuration: number;
    pdt?: string;
    externalBreakId?: string;
  } | null = null;

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i]!;
    const trimmed = line.trim();

    const seqMatch = trimmed.match(MEDIA_SEQUENCE);
    if (seqMatch) {
      mediaSequence = Number(seqMatch[1]);
      continue;
    }

    const pdtMatch = trimmed.match(PROGRAM_DATE_TIME);
    if (pdtMatch) {
      lastPdt = pdtMatch[1]!.trim();
      // continue — PDT applies to the next segment.
    }

    const marker = parseMarkerLine(line, i);
    if (marker?.kind === "daterange" && marker.id) {
      blockBreakId = marker.id;
      // DATERANGE-OUT typically follows the CUE-OUT in the same block; attach it to the open avail.
      if (open && open.externalBreakId === undefined) open.externalBreakId = marker.id;
      continue;
    }
    if (marker?.kind === "cue_out" && !open) {
      open = {
        startLine: i,
        startSequence: mediaSequence,
        measuredDuration: 0,
        ...(marker.duration === undefined ? {} : { declaredDuration: marker.duration }),
        ...(lastPdt === undefined ? {} : { pdt: lastPdt }),
        // Covers a producer that emits DATERANGE before CUE-OUT within the same block.
        ...(blockBreakId === undefined ? {} : { externalBreakId: blockBreakId }),
      };
      continue;
    }
    if (marker?.kind === "cue_in" && open) {
      const declared = open.declaredDuration;
      avails.push({
        opportunityId: `opp_${channelId}_${open.startSequence}`,
        channelId,
        ...(open.externalBreakId === undefined ? {} : { externalBreakId: open.externalBreakId }),
        duration: declared ?? open.measuredDuration,
        ...(declared === undefined ? {} : { declaredDuration: declared }),
        measuredDuration: open.measuredDuration,
        startSequence: open.startSequence,
        ...(open.pdt === undefined ? {} : { programDateTime: open.pdt }),
        startLine: open.startLine,
        endLine: i,
      });
      open = null;
      continue;
    }

    // Count segments (an #EXTINF followed by a URI advances the media sequence).
    const infMatch = trimmed.match(EXTINF);
    if (infMatch) {
      const segDur = Number(infMatch[1]);
      if (open) open.measuredDuration += segDur;
      // The URI line follows; advance the sequence when we consume it below.
    } else if (trimmed.length > 0 && !trimmed.startsWith("#")) {
      // A media segment URI line — the tag block ended, so a DATERANGE id does not carry over.
      mediaSequence += 1;
      blockBreakId = undefined;
    }
  }

  return avails;
}
